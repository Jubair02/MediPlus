    50→/** Compact relative time for Q&A meta lines ("3 days ago") */
    51→function timeAgo(iso: string): string {
    52→  const s = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 1000))
    53→  if (s < 60) return 'just now'
    54→  const m = Math.floor(s / 60)
    55→  if (m < 60) return `${m} minute${m === 1 ? '' : 's'} ago`
    56→  const h = Math.floor(m / 60)
    57→  if (h < 24) return `${h} hour${h === 1 ? '' : 's'} ago`
    58→  const d = Math.floor(h / 24)
    59→  if (d < 30) return `${d} day${d === 1 ? '' : 's'} ago`
    60→  const mo = Math.floor(d / 30)
    61→  if (mo < 12) return `${mo} month${mo === 1 ? '' : 's'} ago`
    62→  const y = Math.floor(mo / 12)
    63→  return `${y} year${y === 1 ? '' : 's'} ago`
    64→}

   165→      expiryChip = { label: `${months} months left`, className: 'bg-amber-100 text-amber-700', title: 'Short-dated batch — discounted freshness' }

   167→      expiryChip = { label: `${months} months left`, className: 'bg-emerald-100 text-emerald-700', title: 'Plenty of shelf life' }

   170→

   172→    setQty((q) => Math.min(Math.max(1, q + delta), maxQty))

   193→            <FileWarning className="size-3.5" aria-hidden="true" />
   194→            Prescription required

   201→        <h2 className="text-xl font-bold leading-tight tracking-tight">{m.name}</h2>
   202→        <p className="mt-1 text-sm text-muted-foreground">
   203→          {[m.brand, m.manufacturer].filter(Boolean).join(' · ') || '—'}
   204→        </p>
   205→        {m.genericName && (
   206→          <p className="mt-0.5 text-sm text-muted-foreground">
   207→            Generic: <span className="font-medium text-foreground">{m.genericName}</span>

   214→            <span className="line-through">{fmtBDT(m.price)}</span>{' '}

   324→/**
   325→ * Product Q&A — public answered threads (chat bubbles) + the viewer's own
   326→ * pending questions. Guests are prompted to sign in before asking.
   327→ */
   328→function QASection({ medicineId }: { medicineId: string }) {
   329→  const user = useAppStore((s) => s.user)
   330→  const setAuthOpen = useAppStore((s) => s.setAuthOpen)
   331→  const [questions, setQuestions] = useState<MedicineQA[] | null>(null)
   332→  const [loading, setLoading] = useState(true)
   333→  const [failed, setFailed] = useState(false)
   334→  const [composerOpen, setComposerOpen] = useState(false)
   335→  const [draft, setDraft] = useState('')
   336→  const [submitting, setSubmitting] = useState(false)
   337→  const [reloadKey, setReloadKey] = useState(0)
   338→  // Per-question in-flight id — only the clicked Helpful button spins/disables
   339→  const [votingId, setVotingId] = useState<string | null>(null)
   340→
   341→  // Auth OPTIONAL on this GET — sending the token also surfaces own PENDING rows
   342→  useEffect(() => {
   343→    const ac = new AbortController()
   344→    api<{ questions: MedicineQA[] }>(
   345→      `/api/questions?medicineId=${encodeURIComponent(medicineId)}`,
   346→      { signal: ac.signal }
   347→    )
   348→      .then((d) => {
   349→        if (ac.signal.aborted) return
   350→        setQuestions(d.questions)
   351→        setFailed(false)
   352→      })
   353→      .catch((err) => {
   354→        if (ac.signal.aborted) return
   355→        setFailed(err instanceof Error)
   356→      })
   357→      .finally(() => {
   358→        if (!ac.signal.aborted) setLoading(false)
   359→      })
   360→    return () => ac.abort()
   361→  }, [medicineId, user?.id, reloadKey])
   362→
   363→  const answered = (questions ?? []).filter((q) => q.status === 'ANSWERED')
   364→  const pendingMine = (questions ?? []).filter((q) => q.status === 'PENDING')
   365→  const draftLen = draft.trim().length
   366→  const canSubmit = draftLen >= MIN_QUESTION && draftLen <= MAX_QUESTION && !submitting
   367→
   368→  const submitQuestion = async () => {
   369→    if (!canSubmit) return
   370→    setSubmitting(true)
   371→    try {
   372→      const d = await api<{ question: MedicineQA }>('/api/questions', {
   373→        method: 'POST',
   374→        body: { medicineId, question: draft.trim() },
   375→      })
   376→      setQuestions((prev) => [d.question, ...(prev ?? [])])
   377→      setComposerOpen(false)
   378→      setDraft('')
   379→      toast.success('Question submitted — our pharmacist will answer soon')
   380→    } catch (err) {
   381→      toast.error(err instanceof Error ? err.message : 'Failed to submit question')
   382→    } finally {
   383→      setSubmitting(false)
   384→    }
   385→  }
   386→
   387→  const onAskClick = () => {
   388→    if (!user) {
   389→      setAuthOpen(true)
   390→      return
   391→    }
   392→    setComposerOpen((o) => !o)
   393→  }
   394→
   395→  /** Optimistic helpful-vote toggle; guests get the AuthModal, never a request */
   396→  const toggleHelpful = async (q: MedicineQA) => {
   397→    if (!user) {
   398→      setAuthOpen(true)
   399→      return
   400→    }
   401→    if (votingId) return // one vote request at a time keeps rollback predictable
   402→    const snapshot = questions
   403→    const nextVoted = !(q.hasVoted ?? false)
   404→    setQuestions((list) =>
   405→      (list ?? []).map((row) =>
   406→        row.id === q.id
   407→          ? {
   408→              ...row,
   409→              hasVoted: nextVoted,
   410→              helpfulCount: Math.max(0, (row.helpfulCount ?? 0) + (nextVoted ? 1 : -1)),
   411→            }
   412→          : row
   413→      )
   414→    )
   415→    setVotingId(q.id)
   416→    try {
   417→      const d = await api<{ questionId: string; helpfulCount: number; voted: boolean }>(
   418→        '/api/questions/helpful',
   419→        { method: 'POST', body: { questionId: q.id } }
   420→      )
   421→      // Reconcile with server truth
   422→      setQuestions((list) =>
   423→        (list ?? []).map((row) =>
   424→          row.id === d.questionId ? { ...row, helpfulCount: d.helpfulCount, hasVoted: d.voted } : row
   425→        )
   426→      )
   427→    } catch (err) {
   428→      const message = err instanceof Error ? err.message : 'Failed to record your vote'
   429→      if (/no longer accepting|only answered/i.test(message)) {
   430→        // 400 — question was likely reverted to pending; refetch for fresh state
   431→        toast.error('This question is no longer accepting votes')
   432→        setReloadKey((k) => k + 1)
   433→      } else {
   434→        toast.error(message)
   435→        setQuestions(snapshot) // rollback the optimistic toggle
   436→      }
   437→    } finally {
   438→      setVotingId(null)
   439→    }
   440→  }
   441→
   442→  return (
   443→    <section aria-label="Questions and answers" className="space-y-4">
   444→      <div className="flex flex-wrap items-center gap-2">
   445→        <h3 className="flex items-center gap-2 text-base font-bold tracking-tight">
   446→          <MessageCircleQuestion className="size-5 text-primary" aria-hidden="true" />
   447→          Questions &amp; answers
   448→          {answered.length > 0 && (
   449→            <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary">
   450→              {answered.length}
   451→            </span>
   452→          )}
   453→        </h3>
   454→        <Button
   455→          variant="outline"
   456→          size="sm"
   457→          className="ml-auto h-9 gap-1.5 rounded-lg"
   458→          onClick={onAskClick}
   459→        >
   460→          <MessageSquarePlus className="size-4" aria-hidden="true" />
   461→          Ask a question
   462→        </Button>
   463→      </div>
   464→
   465→      {loading ? (
   466→        /* 2 gray bubble rows while fetching (pulse + shimmer sweep) */
   467→        <div className="space-y-3" aria-hidden="true">
   468→          {[0, 1].map((i) => (
   469→            <div key={i} className="space-y-2">
   470→              <div className="flex items-start gap-2.5">
   471→                <Skeleton className="relative size-7 shrink-0 overflow-hidden rounded-lg">
   472→                  <Shimmer />
   473→                </Skeleton>
   474→                <Skeleton
   475→                  className={cn('relative flex-1 overflow-hidden rounded-xl', i === 0 ? 'h-9' : 'h-7')}
   476→                >
   477→                  <Shimmer />
   478→                </Skeleton>
   479→              </div>
   480→              <div className="flex items-start gap-2.5 pl-9">
   481→                <Skeleton className="relative size-7 shrink-0 overflow-hidden rounded-lg">
   482→                  <Shimmer />
   483→                </Skeleton>
   484→                <Skeleton className="relative h-12 flex-1 overflow-hidden rounded-xl">
   485→                  <Shimmer />
   486→                </Skeleton>
   487→              </div>
   488→            </div>
   489→          ))}
   490→        </div>
   491→      ) : failed ? (
   492→        <div className="flex flex-col items-center gap-2 rounded-xl border border-dashed p-6 text-center">
   493→          <p className="text-sm text-muted-foreground">Couldn&apos;t load questions.</p>
   494→          <Button
   495→            variant="outline"
   496→            size="sm"
   497→            className="h-9 rounded-lg"
   498→            onClick={() => {
   499→              setLoading(true)
   500→              setReloadKey((k) => k + 1)
   501→            }}
   502→          >
   503→            Try again
   504→          </Button>
   505→        </div>
   506→      ) : (
   507→        <>
   508→          {/* Own pending questions */}
   509→          {pendingMine.length > 0 && (
   510→            <div className="space-y-2">
   511→              <p className="flex items-center gap-1.5 text-xs font-semibold text-amber-700 dark:text-amber-400">
   512→                <Clock3 className="size-3.5" aria-hidden="true" />
   513→                Your question (pending review)
   514→                <span className="rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-semibold text-amber-800 dark:bg-amber-950/60 dark:text-amber-300">
   515→                  Awaiting answer
   516→                </span>
   517→              </p>
   518→              {pendingMine.map((q, i) => (
   519→                <motion.div
   520→                  key={q.id}
   521→                  initial={{ opacity: 0, y: 8 }}
   522→                  animate={{ opacity: 1, y: 0 }}
   523→                  transition={{ duration: 0.25, delay: Math.min(i * 0.04, 0.2), ease: 'easeOut' }}
   524→                  className="rounded-xl border border-dashed border-amber-200 bg-amber-50/50 p-3 dark:border-amber-900/70 dark:bg-amber-950/20"
   525→                >
   526→                  <p className="text-sm leading-snug text-muted-foreground">{q.question}</p>
   527→                  <p className="mt-1 text-[11px] text-muted-foreground">
   528→                    {q.askedByName} · {timeAgo(q.createdAt)}
   529→                  </p>
   530→                </motion.div>
   531→              ))}
   532→            </div>
   533→          )}
   534→
   535→          {/* Answered threads */}
   536→          {answered.length > 0 ? (
   537→            <div
   538→              className={cn(
   539→                'space-y-4',
   540→                answered.length > 4 && 'max-h-80 overflow-y-auto pr-1 scrollbar-thin'
   541→              )}
   542→            >
   543→              {answered.map((q, i) => (
   544→                <motion.div
   545→                  key={q.id}
   546→                  initial={{ opacity: 0, y: 8 }}
   547→                  animate={{ opacity: 1, y: 0 }}
   548→                  transition={{ duration: 0.25, delay: Math.min(i * 0.05, 0.3), ease: 'easeOut' }}
   549→                  className="space-y-2"
   550→                >
   551→                  {/* Question row */}
   552→                  <div className="flex items-start gap-2.5">
   553→                    <span
   554→                      className="flex size-7 shrink-0 items-center justify-center rounded-lg bg-amber-500/15 text-xs font-bold text-amber-700 dark:text-amber-400"
   555→                      aria-hidden="true"
   556→                    >
   557→                      Q
   558→                    </span>
   559→                    <div className="min-w-0 flex-1">
   560→                      <p className="text-sm font-medium leading-snug">{q.question}</p>
   561→                      <p className="mt-0.5 text-[11px] text-muted-foreground">
   562→                        {q.askedByName} · {timeAgo(q.createdAt)}
   563→                      </p>
   564→                    </div>
   565→                  </div>
   566→                  {/* Answer row — indented, pharmacist chip, helpful votes */}
   567→                  {q.answer && (
   568→                    <div className="flex items-start gap-2.5 pl-9">
   569→                      <span
   570→                        className="flex size-7 shrink-0 items-center justify-center rounded-lg bg-primary/15 text-xs font-bold text-primary"
   571→                        aria-hidden="true"
   572→                      >
   573→                        A
   574→                      </span>
   575→                      <div
   576→                        role="group"
   577→                        aria-label={
   578→                          `Answer from ${q.answerByName ?? 'Pharmacist'}` +
   579→                          ((q.helpfulCount ?? 0) > 0
   580→                            ? `, marked helpful by ${q.helpfulCount} ${q.helpfulCount === 1 ? 'person' : 'people'}`
   581→                            : '')
   582→                        }
   583→                        className="min-w-0 flex-1 rounded-xl rounded-tl-sm border bg-card/60 p-3"
   584→                      >
   585→                        <p className="inline-flex items-center gap-1 text-[11px] font-semibold text-primary">
   586→                          <BadgeCheck className="size-3.5" aria-hidden="true" />
   587→                          {q.answerByName ?? 'Pharmacist'}
   588→                        </p>
   589→                        <p className="mt-1 text-sm leading-relaxed text-muted-foreground">{q.answer}</p>
   590→                        <HelpfulButton
   591→                          question={q}
   592→                          busy={votingId === q.id}
   593→                          onToggle={() => void toggleHelpful(q)}
   594→                        />
   595→                      </div>
   596→                    </div>
   597→                  )}
   598→                </motion.div>
   599→              ))}
   600→            </div>
   601→          ) : (
   602→            pendingMine.length === 0 && (
   603→              <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed p-6 text-center">
   604→                <span className="flex size-11 items-center justify-center rounded-full bg-primary/10">
   605→                  <MessageCircleQuestion className="size-5 text-primary" aria-hidden="true" />
   606→                </span>
   607→                <p className="text-sm font-semibold">No questions yet</p>
   608→                <p className="text-xs text-muted-foreground">
   609→                  Be the first to ask about this medicine.
   610→                </p>
   611→              </div>
   612→            )
   613→          )}
   614→
   615→          {/* Inline composer (signed-in only — guests get the auth modal) */}
   616→          {composerOpen && user && (
   617→            <motion.div
   618→              initial={{ opacity: 0, y: 6 }}
   619→              animate={{ opacity: 1, y: 0 }}
   620→              transition={{ duration: 0.2, ease: 'easeOut' }}
   621→              className="rounded-xl border bg-card/50 p-3"
   622→            >
   623→              <Textarea
   624→                value={draft}
   625→                maxLength={MAX_QUESTION}
   626→                onChange={(e) => setDraft(e.target.value)}
   627→                placeholder="Ask about dosage, usage, interactions…"
   628→                aria-label="Your question"
   629→                className="min-h-20 resize-none focus-visible:ring-primary/30"
   630→              />
   631→              {draftLen > 0 && draftLen < MIN_QUESTION && (
   632→                <p className="mt-1 text-[11px] text-amber-600">
   633→                  Question must be at least {MIN_QUESTION} characters.
   634→                </p>
   635→              )}
   636→              <div className="mt-2 flex items-center justify-between gap-2">
   637→                <span
   638→                  className={cn(
   639→                    'text-[11px] tabular-nums',
   640→                    draftLen > 540 ? 'font-medium text-amber-600' : 'text-muted-foreground'
   641→                  )}
   642→                  aria-live="polite"
   643→                >
   644→                  {draft.length} / {MAX_QUESTION}
   645→                </span>
   646→                <div className="flex gap-2">
   647→                  <Button
   648→                    variant="ghost"
   649→                    size="sm"
   650→                    className="h-9 rounded-lg"
   651→                    disabled={submitting}
   652→                    onClick={() => {
   653→                      setComposerOpen(false)
   654→                      setDraft('')
   655→                    }}
   656→                  >
   657→                    Cancel
   658→                  </Button>
   659→                  <Button
   660→                    size="sm"
   661→                    className="h-9 gap-1.5 rounded-lg"
   662→                    disabled={!canSubmit}
   663→                    onClick={() => void submitQuestion()}
   664→                  >
   665→                    {submitting && <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />}
   666→                    {submitting ? 'Submitting…' : 'Submit question'}
   667→                  </Button>
   668→                </div>
   669→              </div>
   670→            </motion.div>
   671→          )}
   672→        </>
   673→      )}
   674→    </section>
   675→  )
   676→}
