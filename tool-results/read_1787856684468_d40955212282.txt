    44→/** Shared data source for the reports view and the export-report CSV (same shape as GET ?resource=reports). */
    45→async function reportData(days: number) {
    46→  const notCancelled = { not: 'CANCELLED' }
    47→  const start = new Date()
    48→  start.setHours(0, 0, 0, 0)
    49→  start.setDate(start.getDate() - (days - 1))
    50→
    51→  const [orders, orderItems, lowStock] = await Promise.all([
    52→    db.order.findMany({
    53→      where: { status: notCancelled, createdAt: { gte: start } },
    54→      select: { createdAt: true, total: true },
    55→    }),
    56→    db.orderItem.findMany({
    57→      where: { order: { status: notCancelled } },
    58→      select: { name: true, price: true, quantity: true, medicine: { select: { category: { select: { name: true } } } } },
    59→    }),
    60→    db.medicine.findMany({
    61→      where: { status: 'ACTIVE', stock: { lte: 10 } },
    62→      include: { category: true },
    63→      orderBy: { stock: 'asc' },
    64→      take: 10,
    65→    }),
    66→  ])
    67→
    68→  const byDay = new Map<string, { revenue: number; orders: number }>()
    69→  for (let i = 0; i < days; i++) {
    70→    const d = new Date(start)
    71→    d.setDate(start.getDate() + i)
    72→    byDay.set(dayKey(d), { revenue: 0, orders: 0 })
    73→  }
    74→  for (const o of orders) {
    75→    const entry = byDay.get(dayKey(o.createdAt))
    76→    if (entry) {
    77→      entry.revenue = round2(entry.revenue + o.total)
    78→      entry.orders += 1
    79→    }
    80→  }
    81→  const salesByDay = Array.from(byDay.entries()).map(([key, v]) => {
    82→    const [y, m, d] = key.split('-').map(Number)
    83→    const date = new Date(y, m - 1, d)
    84→    return {
    85→      date: date.toLocaleDateString('en-GB', { day: 'numeric', month: 'short' }),
    86→      revenue: v.revenue,
    87→      orders: v.orders,
    88→    }
    89→  })
    90→
    91→  const catMap = new Map<string, { qty: number; revenue: number }>()
    92→  const medMap = new Map<string, { qty: number; revenue: number }>()
    93→  for (const it of orderItems) {
    94→    const revenue = round2(it.price * it.quantity)
    95→    const catName = it.medicine?.category?.name ?? 'Uncategorized'
    96→    const cat = catMap.get(catName) ?? { qty: 0, revenue: 0 }
    97→    cat.qty += it.quantity
    98→    cat.revenue = round2(cat.revenue + revenue)
    99→    catMap.set(catName, cat)
   100→    const med = medMap.get(it.name) ?? { qty: 0, revenue: 0 }
   101→    med.qty += it.quantity
   102→    med.revenue = round2(med.revenue + revenue)
   103→    medMap.set(it.name, med)
   104→  }
   105→  const categorySales = Array.from(catMap.entries())
   106→    .map(([category, v]) => ({ category, qty: v.qty, revenue: v.revenue }))
   107→    .sort((a, b) => b.revenue - a.revenue)
   108→  const topMedicines = Array.from(medMap.entries())
   109→    .map(([name, v]) => ({ name, qty: v.qty, revenue: v.revenue }))
   110→    .sort((a, b) => b.qty - a.qty)
   111→    .slice(0, 5)
   112→
   113→  return { salesByDay, categorySales, topMedicines, lowStock }
   114→}

   391→/**
   392→ * PUT /api/admin
   393→ *  {action:'update-user', id, status?, role?}
   394→ *  {action:'create-medicine'|'update-medicine'|'delete-medicine', ...}
   395→ *  {action:'create-category'|'update-category'|'delete-category', ...}
   396→ *  {action:'update-order', id, status?, deliveryStaffId?}
   397→ *  {action:'create-staff', data:{name,email,password,phone?,role}}
   398→ */
   399→export async function PUT(request: Request) {
   400→  try {
   401→    const user = await requireRole(request, ['ADMIN'])
   402→    if (user instanceof Response) return user
   403→    const body = await readJson(request)
   404→    if (!body) return badRequest('Invalid request body')
   405→    const action = optStr(body.action)
   406→
   407→    // ---- users ----
   408→    if (action === 'update-user') {
   409→      const id = optStr(body.id)
   410→      if (!id) return badRequest('User id is required')
   411→      const target = await db.user.findUnique({ where: { id } })
   412→      if (!target) return notFound('User not found')
   413→      const status = optStr(body.status)
   414→      const role = optStr(body.role)
   415→      if (status && status !== 'ACTIVE' && status !== 'INACTIVE') return badRequest('Status must be ACTIVE or INACTIVE')
   416→      if (role && !ROLES.includes(role)) return badRequest('Invalid role')
   417→      if (target.id === user.id) {
   418→        if (role && role !== target.role) return badRequest('You cannot change your own role')
   419→        if (status && status !== target.status) return badRequest('You cannot deactivate your own account')
   420→      }
   421→      const updated = await db.user.update({
   422→        where: { id },
   423→        data: {
   424→          ...(status ? { status } : {}),
   425→          ...(role ? { role } : {}),
   426→        },
   427→      })
   428→      return Response.json({ user: publicUser(updated) })
   429→    }
   430→
   431→    if (action === 'create-staff') {
   432→      const data = (typeof body.data === 'object' && body.data !== null ? body.data : {}) as Record<string, unknown>
   433→      const name = optStr(data.name)?.trim()
   434→      const email = optStr(data.email)?.trim().toLowerCase()
   435→      const password = optStr(data.password)
   436→      const phone = optStr(data.phone)?.trim()
   437→      const role = optStr(data.role)
   438→      if (!name) return badRequest('Name is required')
   439→      if (!email) return badRequest('Email is required')
   440→      if (!password || password.length < 6) return badRequest('Password must be at least 6 characters')
   441→      if (role !== 'PHARMACIST' && role !== 'DELIVERY') return badRequest('Role must be PHARMACIST or DELIVERY')
   442→      const existing = await db.user.findUnique({ where: { email } })
   443→      if (existing) return badRequest('An account with this email already exists')
   444→      const created = await db.user.create({
   445→        data: { name, email, password: hashPassword(password), phone: phone || null, role, status: 'ACTIVE' },
   446→      })
   447→      return Response.json({ user: publicUser(created) }, { status: 201 })
   448→    }
   449→
   450→    // ---- medicines ----
   451→    if (action === 'create-medicine') {
   452→      const parsed = await buildMedicineData(body.data, 'create')
   453→      if ('error' in parsed) return badRequest(parsed.error)
   454→      const medicine = await db.medicine.create({ data: parsed.data, include: { category: true } })
   455→      if (medicine.stock > 0) {
   456→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: medicine.stock, reason: 'MANUAL_EDIT', note: 'Initial stock', userId: user.id }])
   457→      }
   458→      return Response.json({ medicine }, { status: 201 })
   459→    }
   460→
   461→    if (action === 'update-medicine') {
   462→      const id = optStr(body.id)
   463→      if (!id) return badRequest('Medicine id is required')
   464→      const existing = await db.medicine.findUnique({ where: { id } })
   465→      if (!existing) return notFound('Medicine not found')
   466→      const parsed = await buildMedicineData(body.data, 'update')
   467→      if ('error' in parsed) return badRequest(parsed.error)
   468→      const newStock = typeof parsed.data.stock === 'number' ? parsed.data.stock : undefined
   469→      const medicine = await db.medicine.update({ where: { id }, data: parsed.data, include: { category: true } })
   470→      if (newStock !== undefined && newStock !== existing.stock) {
   471→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: newStock - existing.stock, reason: 'MANUAL_EDIT', note: 'Manual stock update', userId: user.id }])
   472→      }
   473→      return Response.json({ medicine })
   474→    }
   475→
   476→    if (action === 'delete-medicine') {
   477→      const id = optStr(body.id)
   478→      if (!id) return badRequest('Medicine id is required')
   479→      const existing = await db.medicine.findUnique({ where: { id } })
   480→      if (!existing) return notFound('Medicine not found')
   481→      await db.medicine.update({ where: { id }, data: { status: 'INACTIVE' } })
   482→      await db.cartItem.deleteMany({ where: { medicineId: id } })
   483→      return Response.json({ ok: true })
   484→    }
   485→
   486→    // ---- categories ----
   487→    if (action === 'create-category') {
   488→      const parsed = parseCategoryInput(body.data, 'create')
   489→      if ('error' in parsed) return badRequest(parsed.error)
   490→      const name = parsed.data.name as string
   491→      const existing = await db.category.findFirst({ where: { name } })
   492→      if (existing) return badRequest('A category with this name already exists')
   493→      let slug = slugify(name)
   494→      const slugTaken = await db.category.findUnique({ where: { slug } })
   495→      if (slugTaken) slug = `${slug}-${Date.now().toString(36)}`
   496→      const category = await db.category.create({ data: { name, slug, description: parsed.data.description ?? null } })
   497→      return Response.json({ category }, { status: 201 })
   498→    }
   499→
   500→    if (action === 'update-category') {
   501→      const id = optStr(body.id)
   502→      if (!id) return badRequest('Category id is required')
   503→      const existing = await db.category.findUnique({ where: { id } })
   504→      if (!existing) return notFound('Category not found')
   505→      const parsed = parseCategoryInput(body.data, 'update')
   506→      if ('error' in parsed) return badRequest(parsed.error)
   507→      const name = parsed.data.name ?? existing.name
   508→      if (parsed.data.name) {
   509→        const clash = await db.category.findFirst({ where: { name: parsed.data.name, NOT: { id } } })
   510→        if (clash) return badRequest('A category with this name already exists')
   511→      }
   512→      const category = await db.category.update({
   513→        where: { id },
   514→        data: { name, slug: slugify(name), description: parsed.data.description !== undefined ? parsed.data.description : existing.description },
   515→      })
   516→      return Response.json({ category })
   517→    }
   518→
   519→    if (action === 'delete-category') {
   520→      const id = optStr(body.id)
   521→      if (!id) return badRequest('Category id is required')
   522→      const existing = await db.category.findUnique({ where: { id } })
   523→      if (!existing) return notFound('Category not found')
   524→      await db.category.delete({ where: { id } })
   525→      return Response.json({ ok: true })
   526→    }
   527→
   528→    // ---- orders ----
   529→    if (action === 'update-order') {
   530→      const id = optStr(body.id)
   531→      if (!id) return badRequest('Order id is required')
   532→      const order = await db.order.findUnique({
   533→        where: { id },
   534→        include: { ...orderInclude, items: { include: { medicine: true } } },
   535→      })
   536→      if (!order) return notFound('Order not found')
   537→
   538→      const status = optStr(body.status)
   539→      const deliveryStaffId = optStr(body.deliveryStaffId)
   540→
   541→      if (status && !ORDER_STATUSES.includes(status)) return badRequest('Invalid status')
   542→      if (deliveryStaffId) {
   543→        const staff = await db.user.findUnique({ where: { id: deliveryStaffId } })
   544→        if (!staff || staff.role !== 'DELIVERY') return badRequest('Selected user is not delivery staff')
   545→      }
   546→
   547→      if (status && status !== order.status) {
   548→        if (status === 'CONFIRMED' && ['PENDING', 'PRESCRIPTION_REVIEW'].includes(order.status)) {
   549→          for (const item of order.items) {
   550→            if (!item.medicine || item.medicine.status !== 'ACTIVE' || item.medicine.stock < item.quantity) {
   551→              return badRequest(`Insufficient stock for ${item.name}`)
   552→            }
   553→          }
   554→          await db.$transaction(async (tx) => {
   555→            const movements: StockMovementEntry[] = []
   556→            for (const item of order.items) {
   557→              if (item.medicineId) {
   558→                await tx.medicine.update({ where: { id: item.medicineId }, data: { stock: { decrement: item.quantity } } })
   559→                movements.push({
   560→                  medicineId: item.medicineId,
   561→                  delta: -item.quantity,
   562→                  reason: 'ORDER_CONFIRM',
   563→                  note: `Order ${order.orderNo} confirmed`,
   564→                  userId: user.id,
   565→                })
   566→              }
   567→            }
   568→            await recordStockMovements(tx, movements)
   569→            const existingPayment = await tx.payment.findUnique({ where: { orderId: order.id } })
   570→            if (!existingPayment) {
   571→              await tx.payment.create({
   572→                data: {
   573→                  orderId: order.id,
   574→                  method: order.paymentMethod,
   575→                  status: order.paymentStatus === 'PAID' ? 'PAID' : 'PENDING',
   576→                  amount: order.total,
   577→                },
   578→              })
   579→            }
   580→          })
   581→        }
   582→        if (status === 'CANCELLED' && ['CONFIRMED', 'PROCESSING', 'OUT_FOR_DELIVERY'].includes(order.status)) {
   583→          const movements: StockMovementEntry[] = []
   584→          await db.$transaction(async (tx) => {
   585→            for (const item of order.items) {
   586→              if (item.medicineId) {
   587→                await tx.medicine.update({ where: { id: item.medicineId }, data: { stock: { increment: item.quantity } } })
   588→                movements.push({
   589→                  medicineId: item.medicineId,
   590→                  delta: item.quantity,
   591→                  reason: 'ORDER_CANCEL',
   592→                  note: `Order ${order.orderNo} cancelled — stock restored`,
   593→                  userId: user.id,
   594→                })
   595→              }
   596→            }
   597→            await recordStockMovements(tx, movements)
   598→          })
   599→        }
   600→        if (status === 'CANCELLED' && order.paymentStatus === 'PAID') {
   601→          if (order.payment) {
   602→            await db.payment.update({ where: { id: order.payment.id }, data: { status: 'REFUNDED' } })
   603→          }
   604→        }
   605→      }
   606→
   607→      const wasPaid = order.paymentStatus === 'PAID'
   608→      const updated = await db.order.update({
   609→        where: { id },
   610→        data: {
   611→          ...(status ? { status, statusNote: 'Updated by admin' } : {}),
   612→          ...(status === 'CANCELLED' && wasPaid ? { paymentStatus: 'REFUNDED' } : {}),
   613→          ...(deliveryStaffId ? { deliveryStaffId } : {}),
   614→        },
   615→      })
   616→
   617→      if (status && status !== order.status) {
   618→        const label = ORDER_STATUS_LABELS[status as OrderStatus] ?? status
   619→        await notify(order.userId, label, `Order ${order.orderNo} status updated to ${label}.`)
   620→      }
   621→
   622→      const full = await db.order.findUnique({ where: { id: updated.id }, include: orderInclude })
   623→      if (!full) return notFound('Order not found')
   624→      return Response.json({ order: parseOrder(full, { prescriptionImage: true }) })
   625→    }
   626→
   627→    return badRequest('Invalid action')
   628→  } catch (e) {
   629→    return serverError(e)
   630→  }
   631→}
