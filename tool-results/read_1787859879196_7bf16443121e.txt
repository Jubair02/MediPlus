   211→   211→  }
   212→   212→
   213→   213→  // ---------- loading skeleton ----------
   214→   214→  if (!ready) {
   215→   215→    return (
   216→   216→      <div className="mx-auto max-w-6xl px-4 py-8 lg:px-6">
   217→   217→        <Skeleton className="mb-6 h-8 w-56" />
   218→   218→        <div className="grid gap-6 lg:grid-cols-[1fr_22rem]">
   219→   219→          <div className="space-y-4">
   220→   220→            {Array.from({ length: 3 }).map((_, i) => (
   221→   221→              <Skeleton key={i} className="h-40 rounded-xl" />

   239→   239→          Your cart is empty — add some medicines first.

   242→   242→          Browse medicines

   247→   247→
   248→   248→  return (
   249→   249→    <div className="mx-auto max-w-6xl px-4 pb-16 pt-6 lg:px-6">

   251→   251→
   252→   252→      <div className="mt-6 grid items-start gap-6 lg:grid-cols-[1fr_22rem]">
   253→   253→        {/* ---------- Left column ---------- */}
   254→   254→        <div className="space-y-6">
