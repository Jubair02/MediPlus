   101→/** Low-stock ACTIVE medicines with 30-day sales velocity → reorder suggestions. */
   102→async function restockSuggestions() {
   103→  const since = new Date(Date.now() - RESTOCK_WINDOW_DAYS * 24 * 60 * 60 * 1000)
   104→  const medicines = await db.medicine.findMany({
   105→    where: { status: 'ACTIVE', stock: { lte: LOW_STOCK_THRESHOLD } },
   106→    orderBy: { stock: 'asc' },
   107→  })
   108→  const ids = medicines.map((m) => m.id)
   109→  const soldRows = ids.length
   110→    ? await db.orderItem.findMany({
   111→        where: {
   112→          medicineId: { in: ids },
   113→          order: { status: { notIn: ['CANCELLED', 'FAILED'] }, createdAt: { gte: since } },
   114→        },
   115→        select: { medicineId: true, quantity: true },
   116→      })
   117→    : []
   118→  const soldByMed = new Map<string, number>()
   119→  for (const row of soldRows) {
   120→    if (row.medicineId) soldByMed.set(row.medicineId, (soldByMed.get(row.medicineId) ?? 0) + row.quantity)
   121→  }
   122→  // Round 10 — total qty across each medicine's OPEN (ORDERED) purchase orders
   123→  const openPoGroups = ids.length
   124→    ? await db.purchaseOrder.groupBy({ by: ['medicineId'], where: { medicineId: { in: ids }, status: 'ORDERED' }, _sum: { qty: true } })
   125→    : []
   126→  const openPoByMed = new Map(openPoGroups.map((g) => [g.medicineId, g._sum.qty ?? 0]))
   127→
   128→  const suggestions = medicines.map((m) => {
   129→    const soldLast30 = soldByMed.get(m.id) ?? 0
   130→    const avgDaily = round1(soldLast30 / RESTOCK_WINDOW_DAYS)
   131→    let daysLeft: number | null
   132→    if (m.stock <= 0) daysLeft = 0
   133→    else if (avgDaily <= 0) daysLeft = null
   134→    else daysLeft = round1(m.stock / avgDaily)
   135→    const base = Math.max(30, Math.ceil(soldLast30 * 1.5)) - m.stock
   136→    let suggestedQty = Math.ceil(Math.max(base, 10) / 10) * 10
   137→    if (m.stock <= 0) suggestedQty = Math.max(suggestedQty, 30)
   138→    return {
   139→      id: m.id,
   140→      name: m.name,
   141→      brand: m.brand,
   142→      genericName: m.genericName,
   143→      unit: m.unit,
   144→      image: m.image,
   145→      stock: m.stock,
   146→      lowStockAt: LOW_STOCK_THRESHOLD,
   147→      soldLast30,
   148→      avgDaily,
   149→      daysLeft,
   150→      suggestedQty,
   151→      estValue: round2(suggestedQty * (m.discountPrice ?? m.price)),
   152→      openPoQty: openPoByMed.get(m.id) ?? 0,
   153→    }
   154→  })
   155→  // daysLeft asc (nulls last), then stock asc
   156→  suggestions.sort((a, b) => {
   157→    const an = a.daysLeft === null ? 1 : 0
   158→    const bn = b.daysLeft === null ? 1 : 0
   159→    if (an !== bn) return an - bn
   160→    if (a.daysLeft !== null && b.daysLeft !== null && a.daysLeft !== b.daysLeft) return a.daysLeft - b.daysLeft
   161→    return a.stock - b.stock
   162→  })
   163→  return { suggestions }
   164→}

   166→/**
   167→ * GET /api/pharmacist?resource=stats|prescriptions|medicines|orders|movements|questions|restock-suggestions|purchase-orders|export-restock
   168→ *   resource=medicines       — staff rows carry extraImages (urls sorted asc) + imageCount (Round 11 gallery)
   169→ *   resource=questions       — rows carry edited: true when the text was edited while PENDING (Round 11)
   170→ *   resource=purchase-orders — rows carry supplier / expectedAt (Round 11)
   171→ */
   172→export async function GET(request: Request) {
   173→  try {
   174→    const user = await requireRole(request, ['PHARMACIST'])
   175→    if (user instanceof Response) return user
   176→    const sp = new URL(request.url).searchParams
   177→    const resource = sp.get('resource') || 'stats'
   178→
   179→    if (resource === 'stats') {
   180→      const todayStart = new Date()
   181→      todayStart.setHours(0, 0, 0, 0)
   182→      const in90Days = new Date(Date.now() + 90 * 24 * 60 * 60 * 1000)
   183→      // APPROVED rx whose approval expires within RX_EXPIRY_WARNING_DAYS (expiresAt = reviewedAt + RX_VALIDITY_DAYS ≤ now + 14d,
   184→      // i.e. reviewedAt ≤ now − 76d) — includes already-expired approvals.
   185→      const rxExpiringCutoff = new Date(Date.now() - (RX_VALIDITY_DAYS - RX_EXPIRY_WARNING_DAYS) * 24 * 60 * 60 * 1000)
   186→      const [pendingPrescriptions, approvedToday, rejectedToday, totalMedicines, lowStockCount, prescriptionMedicines, lowStock, expiringSoon, expiringSoonCount, rxExpiringSoon, openPoCount] =
   187→        await Promise.all([
   188→          db.prescription.count({ where: { status: 'PENDING' } }),
   189→          db.prescription.count({ where: { status: 'APPROVED', updatedAt: { gte: todayStart } } }),
   190→          db.prescription.count({ where: { status: 'REJECTED', updatedAt: { gte: todayStart } } }),
   191→          db.medicine.count({ where: { status: 'ACTIVE' } }),
   192→          db.medicine.count({ where: { status: 'ACTIVE', stock: { lte: 10 } } }),
   193→          db.medicine.count({ where: { status: 'ACTIVE', requiresPrescription: true } }),
   194→          db.medicine.findMany({
   195→            where: { status: 'ACTIVE', stock: { lte: 10 } },
   196→            include: { category: true },
   197→            orderBy: { stock: 'asc' },
   198→            take: 10,
   199→          }),
   200→          db.medicine.findMany({
   201→            where: { status: 'ACTIVE', expiryDate: { not: null, lte: in90Days } },
   202→            include: { category: true },
   203→            orderBy: { expiryDate: 'asc' },
   204→            take: 10,
   205→          }),
   206→          db.medicine.count({ where: { status: 'ACTIVE', expiryDate: { not: null, lte: in90Days } } }),
   207→          db.prescription.count({ where: { status: 'APPROVED', reviewedAt: { not: null, lte: rxExpiringCutoff } } }),
   208→          db.purchaseOrder.count({ where: { status: 'ORDERED' } }),
   209→        ])
   210→      return Response.json({
   211→        stats: { pendingPrescriptions, approvedToday, rejectedToday, totalMedicines, lowStockCount, prescriptionMedicines, lowStock, expiringSoon, expiringSoonCount, rxExpiringSoon, openPoCount },
   212→      })
   213→    }
   214→
   215→    if (resource === 'prescriptions') {
   216→      const status = sp.get('status')
   217→      const where = status && RX_STATUSES.includes(status) ? { status } : {}
   218→      const prescriptions = await db.prescription.findMany({
   219→        where,
   220→        include: {
   221→          user: { select: { id: true, name: true, email: true, phone: true } },
   222→          order: { select: { orderNo: true } },
   223→        },
   224→        orderBy: { createdAt: 'desc' },
   225→      })
   226→      return Response.json({
   227→        prescriptions: prescriptions.map((p) => ({
   228→          id: p.id,
   229→          image: p.image,
   230→          note: p.note,
   231→          status: p.status,
   232→          reviewNote: p.reviewNote,
   233→          createdAt: p.createdAt,
   234→          user: p.user,
   235→          orderNo: p.order?.orderNo ?? null,
   236→          ...rxExpiryFields(p.status, p.reviewedAt),
   237→        })),
   238→      })
   239→    }
   240→
   241→    if (resource === 'medicines') {
   242→      const search = sp.get('search')?.trim()
   243→      const medicines = await db.medicine.findMany({
   244→        where: search
   245→          ? {
   246→              OR: [
   247→                { name: { contains: search } },
   248→                { genericName: { contains: search } },
   249→                { brand: { contains: search } },
   250→              ],
   251→            }
   252→          : {},
   253→        include: { category: true, extraImages: { orderBy: { sort: 'asc' } } },
   254→        orderBy: { createdAt: 'desc' },
   255→      })
   256→      return Response.json({ medicines: medicines.map(medicineStaffJson) })
   257→    }
   258→
   259→    if (resource === 'orders') {
   260→      const status = sp.get('status')
   261→      const where = status && ORDER_STATUSES.includes(status) ? { status } : {}
   262→      const orders = await db.order.findMany({
   263→        where,
   264→        include: orderInclude,
   265→        orderBy: { createdAt: 'desc' },
   266→      })
   267→      // PRESCRIPTION_REVIEW first, then newest first
   268→      orders.sort((a, b) => {
   269→        const ap = a.status === 'PRESCRIPTION_REVIEW' ? 0 : 1
   270→        const bp = b.status === 'PRESCRIPTION_REVIEW' ? 0 : 1
   271→        if (ap !== bp) return ap - bp
   272→        return b.createdAt.getTime() - a.createdAt.getTime()
   273→      })
   274→      return Response.json({ orders: orders.map((o) => parseOrder(o, { prescriptionImage: true })) })
   275→    }
   276→
   277→    if (resource === 'movements') {
   278→      const medicineId = sp.get('medicineId')?.trim()
   279→      const take = Math.min(100, Math.max(1, Math.trunc(numParam(sp.get('take')) ?? 50)))
   280→      const movements = await db.stockMovement.findMany({
   281→        where: medicineId ? { medicineId } : {},
   282→        include: {
   283→          medicine: { select: { id: true, name: true, image: true, unit: true } },
   284→          user: { select: { id: true, name: true } },
   285→        },
   286→        orderBy: { createdAt: 'desc' },
   287→        take,
   288→      })
   289→      return Response.json({
   290→        movements: movements.map((m) => ({
   291→          id: m.id,
   292→          delta: m.delta,
   293→          reason: m.reason,
   294→          note: m.note,
   295→          createdAt: m.createdAt,
   296→          medicine: m.medicine,
   297→          user: m.user,
   298→        })),
   299→      })
   300→    }
   301→
   302→    if (resource === 'questions') {
   303→      const status = sp.get('status')
   304→      const st = status && (status === 'ALL' || QA_STATUSES.includes(status)) ? status : 'PENDING'
   305→      const where = st === 'ALL' ? {} : { status: st }
   306→      const [questions, pendingCount, answeredCount, rejectedCount] = await Promise.all([
   307→        db.question.findMany({ where, include: questionInclude, orderBy: { createdAt: 'desc' } }),
   308→        db.question.count({ where: { status: 'PENDING' } }),
   309→        db.question.count({ where: { status: 'ANSWERED' } }),
   310→        db.question.count({ where: { status: 'REJECTED' } }),
   311→      ])
   312→      return Response.json({
   313→        questions: questions.map(questionToJson),
   314→        counts: { PENDING: pendingCount, ANSWERED: answeredCount, REJECTED: rejectedCount },
   315→      })
   316→    }
   317→
   318→    if (resource === 'restock-suggestions') {
   319→      return Response.json(await restockSuggestions())
   320→    }
   321→
   322→    if (resource === 'purchase-orders') {
   323→      const [orders, orderedCount, receivedCount, cancelledCount] = await Promise.all([
   324→        db.purchaseOrder.findMany({ include: poInclude, orderBy: { orderedAt: 'desc' } }),
   325→        db.purchaseOrder.count({ where: { status: 'ORDERED' } }),
   326→        db.purchaseOrder.count({ where: { status: 'RECEIVED' } }),
   327→        db.purchaseOrder.count({ where: { status: 'CANCELLED' } }),
   328→      ])
   329→      return Response.json({
   330→        orders: orders.map(poToRow),
   331→        counts: { ORDERED: orderedCount, RECEIVED: receivedCount, CANCELLED: cancelledCount },
   332→      })
   333→    }
   334→
   335→    if (resource === 'export-restock') {
   336→      const { suggestions } = await restockSuggestions()
   337→      const header = ['id', 'name', 'brand', 'genericName', 'unit', 'image', 'stock', 'lowStockAt', 'soldLast30', 'avgDaily', 'daysLeft', 'suggestedQty', 'estValue']
   338→      const rows = suggestions.map((s) =>
   339→        csvRow([
   340→          s.id,
   341→          s.name,
   342→          s.brand ?? '',
   343→          s.genericName ?? '',
   344→          s.unit,
   345→          s.image ?? '',
   346→          s.stock,
   347→          s.lowStockAt,
   348→          s.soldLast30,
   349→          s.avgDaily,
   350→          s.daysLeft ?? '',
   351→          s.suggestedQty,
   352→          s.estValue,
   353→        ])
   354→      )
   355→      return Response.json({ filename: `restock-${csvDate(new Date())}.csv`, csv: [csvRow(header), ...rows].join('\n') })
   356→    }
   357→
   358→    return badRequest('Unknown resource')
   359→  } catch (e) {
   360→    return serverError(e)
   361→  }
   362→}

   364→/**
   365→ * PUT /api/pharmacist
   366→ *  {action:'review', prescriptionId, decision:'APPROVED'|'REJECTED', reviewNote?}  — sets reviewedAt + RX_REVIEW audit entry
   367→ *  {action:'create-po', medicineId, qty, note?, supplier?, expectedAt?} — create purchase order (PO_CREATE audit entry);
   368→ *      supplier trimmed, empty → null, ≤120 chars else 400 'Supplier must be 120 characters or less'; expectedAt
   369→ *      'YYYY-MM-DD' (local midnight) or ISO string, NaN → 400 'Invalid expected date', before today → 400
   370→ *      'Expected date cannot be in the past'
   371→ *  {action:'receive-po', id, note?}                  — receive PO: stock +qty, StockMovement PO_RECEIVE, audit entry
   372→ *  {action:'cancel-po', id}                          — cancel an ORDERED purchase order (PO_CANCEL audit entry)
   373→ *  {action:'create-medicine', data:{...}}            — data.extraImages (optional, ≤5 data:/http(s): urls) persisted with sort = index
   374→ *  {action:'update-medicine', id, data:{...}}        — data.extraImages present = REPLACE-ALL (empty array clears); absent = untouched
   375→ *  {action:'answer', id, answer}   — answer a Q&A question (re-answer allowed, overwrites)
   376→ *  {action:'reject', id}           — reject a Q&A question (answer/answeredBy/answeredAt cleared)
   377→ */
   378→export async function PUT(request: Request) {
   379→  try {
   380→    const user = await requireRole(request, ['PHARMACIST'])
   381→    if (user instanceof Response) return user
   382→    const body = await readJson(request)
   383→    if (!body) return badRequest('Invalid request body')
   384→    const action = optStr(body.action)
   385→
   386→    if (action === 'review') {
   387→      const prescriptionId = optStr(body.prescriptionId)
   388→      if (!prescriptionId) return badRequest('Prescription id is required')
   389→      const decision = optStr(body.decision)
   390→      if (decision !== 'APPROVED' && decision !== 'REJECTED') return badRequest('Decision must be APPROVED or REJECTED')
   391→
   392→      const prescription = await db.prescription.findUnique({
   393→        where: { id: prescriptionId },
   394→        include: { order: { include: { ...orderInclude, items: { include: { medicine: true } } } } },
   395→      })
   396→      if (!prescription) return notFound('Prescription not found')
   397→      if (prescription.status !== 'PENDING') return badRequest('Already reviewed')
   398→
   399→      const reviewNote = optStr(body.reviewNote)?.trim() || null
   400→      const order = prescription.order
   401→
   402→      if (decision === 'APPROVED' && order) {
   403→        // Check stock for all Rx order items BEFORE approving
   404→        for (const item of order.items) {
   405→          if (!item.medicine) return badRequest(`Insufficient stock for ${item.name}`)
   406→          if (item.medicine.status !== 'ACTIVE' || item.medicine.stock < item.quantity) {
   407→            return badRequest(`Insufficient stock for ${item.name}`)
   408→          }
   409→        }
   410→        await db.$transaction(async (tx) => {
   411→          await tx.prescription.update({
   412→            where: { id: prescription.id },
   413→            data: { status: 'APPROVED', reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   414→          })
   415→          const movements: StockMovementEntry[] = []
   416→          for (const item of order.items) {
   417→            if (item.medicineId) {
   418→              await tx.medicine.update({ where: { id: item.medicineId }, data: { stock: { decrement: item.quantity } } })
   419→              movements.push({
   420→                medicineId: item.medicineId,
   421→                delta: -item.quantity,
   422→                reason: 'RX_APPROVE',
   423→                note: `Order ${order.orderNo} confirmed after prescription approval`,
   424→                userId: user.id,
   425→              })
   426→            }
   427→          }
   428→          await recordStockMovements(tx, movements)
   429→          await tx.order.update({ where: { id: order.id }, data: { status: 'CONFIRMED' } })
   430→          const existingPayment = await tx.payment.findUnique({ where: { orderId: order.id } })
   431→          if (!existingPayment) {
   432→            await tx.payment.create({
   433→              data: {
   434→                orderId: order.id,
   435→                method: order.paymentMethod,
   436→                status: order.paymentStatus === 'PAID' ? 'PAID' : 'PENDING',
   437→                amount: order.total,
   438→              },
   439→            })
   440→          }
   441→          await tx.notification.create({
   442→            data: { userId: order.userId, title: 'Prescription approved', message: 'Your prescription has been approved.' },
   443→          })
   444→          await tx.notification.create({
   445→            data: { userId: order.userId, title: 'Order confirmed', message: `Order ${order.orderNo} has been confirmed and is being processed.` },
   446→          })
   447→          await logAudit(tx, user, {
   448→            action: 'RX_REVIEW',
   449→            entityType: 'PRESCRIPTION',
   450→            entityRef: prescription.id,
   451→            detail: `Prescription ${decision === 'APPROVED' ? 'approved' : 'rejected'}`,
   452→          })
   453→        })
   454→      } else if (decision === 'REJECTED' && order) {
   455→        await db.$transaction(async (tx) => {
   456→          await tx.prescription.update({
   457→            where: { id: prescription.id },
   458→            data: { status: 'REJECTED', reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   459→          })
   460→          await tx.order.update({ where: { id: order.id }, data: { status: 'CANCELLED', statusNote: 'Prescription rejected' } })
   461→          await tx.notification.create({
   462→            data: {
   463→              userId: order.userId,
   464→              title: 'Prescription rejected',
   465→              message: `Your prescription was rejected.${reviewNote ? ` Reason: ${reviewNote}` : ''}`,
   466→            },
   467→          })
   468→          await logAudit(tx, user, {
   469→            action: 'RX_REVIEW',
   470→            entityType: 'PRESCRIPTION',
   471→            entityRef: prescription.id,
   472→            detail: 'Prescription rejected',
   473→          })
   474→        })
   475→      } else {
   476→        await db.prescription.update({
   477→          where: { id: prescription.id },
   478→          data: { status: decision, reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   479→        })
   480→        await logAudit(db, user, {
   481→          action: 'RX_REVIEW',
   482→          entityType: 'PRESCRIPTION',
   483→          entityRef: prescription.id,
   484→          detail: `Prescription ${decision === 'APPROVED' ? 'approved' : 'rejected'}`,
   485→        })
   486→        await notify(
   487→          prescription.userId,
   488→          decision === 'APPROVED' ? 'Prescription approved' : 'Prescription rejected',
   489→          decision === 'APPROVED'
   490→            ? 'Your prescription has been approved.'
   491→            : `Your prescription was rejected.${reviewNote ? ` Reason: ${reviewNote}` : ''}`
   492→        )
   493→      }
   494→
   495→      const updated = await db.prescription.findUnique({
   496→        where: { id: prescription.id },
   497→        include: { order: { select: { orderNo: true } } },
   498→      })
   499→      if (!updated) return notFound('Prescription not found')
   500→      return Response.json({
   501→        prescription: {
   502→          id: updated.id,
   503→          image: updated.image,
   504→          note: updated.note,
   505→          status: updated.status,
   506→          reviewNote: updated.reviewNote,
   507→          createdAt: updated.createdAt,
   508→          orderNo: updated.order?.orderNo ?? null,
   509→        },
   510→      })
   511→    }
   512→
   513→    if (action === 'create-po') {
   514→      const medicineId = optStr(body.medicineId)?.trim()
   515→      if (!medicineId) return badRequest('Medicine id is required')
   516→      const medicine = await db.medicine.findUnique({ where: { id: medicineId } })
   517→      if (!medicine) return notFound('Medicine not found')
   518→      const qty = numOr(body.qty, NaN)
   519→      if (!Number.isInteger(qty) || qty < 1 || qty > PO_MAX_QTY) return badRequest('Quantity must be between 1 and 10000')
   520→      let note: string | null = null
   521→      if (typeof body.note === 'string' && body.note.trim() !== '') {
   522→        note = body.note.trim()
   523→        if (note.length > PO_MAX_NOTE_LENGTH) return badRequest('Note must be 300 characters or less')
   524→      }
   525→      // Round 11 — optional supplier (trimmed, empty → null) + expected delivery date
   526→      let supplier: string | null = null
   527→      if (typeof body.supplier === 'string' && body.supplier.trim() !== '') {
   528→        supplier = body.supplier.trim()
   529→        if (supplier.length > PO_MAX_SUPPLIER_LENGTH) return badRequest('Supplier must be 120 characters or less')
   530→      }
   531→      let expectedAt: Date | null = null
   532→      if (typeof body.expectedAt === 'string' && body.expectedAt.trim() !== '') {
   533→        const raw = body.expectedAt.trim()
   534→        // 'YYYY-MM-DD' is treated as local midnight; anything else parses as an ISO/date string
   535→        const dt = /^\d{4}-\d{2}-\d{2}$/.test(raw)
   536→          ? (() => {
   537→              const [y, m, d] = raw.split('-').map(Number)
   538→              return new Date(y, m - 1, d)
   539→            })()
   540→          : new Date(raw)
   541→        if (Number.isNaN(dt.getTime())) return badRequest('Invalid expected date')
   542→        const todayStart = new Date()
   543→        todayStart.setHours(0, 0, 0, 0)
   544→        if (dt.getTime() < todayStart.getTime()) return badRequest('Expected date cannot be in the past')
   545→        expectedAt = dt
   546→      }
   547→      const created = await db.purchaseOrder.create({
   548→        data: { medicineId: medicine.id, qty, note, supplier, expectedAt, orderedById: user.id },
   549→      })
   550→      await logAudit(db, user, {
   551→        action: 'PO_CREATE',
   552→        entityType: 'PURCHASE_ORDER',
   553→        entityRef: created.id,
   554→        detail: `Ordered ${qty} × ${medicine.name}`,
   555→      })
   556→      const fresh = await db.purchaseOrder.findUnique({ where: { id: created.id }, include: poInclude })
   557→      if (!fresh) return notFound('Purchase order not found')
   558→      return Response.json({ order: poToRow(fresh) })
   559→    }
   560→
   561→    if (action === 'receive-po') {
   562→      const id = optStr(body.id)
   563→      if (!id) return badRequest('Purchase order id is required')
   564→      const po = await db.purchaseOrder.findUnique({
   565→        where: { id },
   566→        include: { medicine: { select: { id: true, name: true } } },
   567→      })
   568→      if (!po) return notFound('Purchase order not found')
   569→      if (po.status !== 'ORDERED') return badRequest('Only ordered purchase orders can be received')
   570→      // body.note replaces the note only when provided as a non-empty string; otherwise the note is kept
   571→      const replacementNote = typeof body.note === 'string' && body.note.trim() !== '' ? body.note.trim() : null
   572→      await db.$transaction(async (tx) => {
   573→        await tx.purchaseOrder.update({
   574→          where: { id: po.id },
   575→          data: {
   576→            status: 'RECEIVED',
   577→            receivedAt: new Date(),
   578→            receivedById: user.id,
   579→            ...(replacementNote !== null ? { note: replacementNote } : {}),
   580→          },
   581→        })
   582→        await tx.medicine.update({ where: { id: po.medicineId }, data: { stock: { increment: po.qty } } })
   583→        await recordStockMovements(tx, [{
   584→          medicineId: po.medicineId,
   585→          delta: po.qty,
   586→          reason: 'PO_RECEIVE',
   587→          note: `PO received — ${po.qty} × ${po.medicine.name}`,
   588→          userId: user.id,
   589→        }])
   590→        await logAudit(tx, user, {
   591→          action: 'PO_RECEIVE',
   592→          entityType: 'PURCHASE_ORDER',
   593→          entityRef: po.id,
   594→          detail: `Received ${po.qty} × ${po.medicine.name}`,
   595→        })
   596→      })
   597→      const fresh = await db.purchaseOrder.findUnique({ where: { id: po.id }, include: poInclude })
   598→      const medicine = await db.medicine.findUnique({ where: { id: po.medicineId }, select: { id: true, stock: true } })
   599→      if (!fresh || !medicine) return notFound('Purchase order not found')
   600→      return Response.json({ order: poToRow(fresh), medicine: { id: medicine.id, stock: medicine.stock } })
   601→    }
   602→
   603→    if (action === 'cancel-po') {
   604→      const id = optStr(body.id)
   605→      if (!id) return badRequest('Purchase order id is required')
   606→      const po = await db.purchaseOrder.findUnique({
   607→        where: { id },
   608→        include: { medicine: { select: { name: true } } },
   609→      })
   610→      if (!po) return notFound('Purchase order not found')
   611→      if (po.status !== 'ORDERED') return badRequest('Only ordered purchase orders can be cancelled')
   612→      await db.purchaseOrder.update({ where: { id: po.id }, data: { status: 'CANCELLED' } })
   613→      await logAudit(db, user, {
   614→        action: 'PO_CANCEL',
   615→        entityType: 'PURCHASE_ORDER',
   616→        entityRef: po.id,
   617→        detail: `Cancelled ${po.qty} × ${po.medicine.name}`,
   618→      })
   619→      const fresh = await db.purchaseOrder.findUnique({ where: { id: po.id }, include: poInclude })
   620→      if (!fresh) return notFound('Purchase order not found')
   621→      return Response.json({ order: poToRow(fresh) })
   622→    }
   623→
   624→    if (action === 'create-medicine') {
   625→      const parsed = await buildMedicineData(body.data, 'create')
   626→      if ('error' in parsed) return badRequest(parsed.error)
   627→      const medicine = await db.medicine.create({
   628→        data: parsed.data,
   629→        include: { category: true, extraImages: { orderBy: { sort: 'asc' } } },
   630→      })
   631→      if (medicine.stock > 0) {
   632→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: medicine.stock, reason: 'MANUAL_EDIT', note: 'Initial stock', userId: user.id }])
   633→      }
   634→      return Response.json({ medicine: medicineStaffJson(medicine) }, { status: 201 })
   635→    }
   636→
   637→    if (action === 'update-medicine') {
   638→      const id = optStr(body.id)
   639→      if (!id) return badRequest('Medicine id is required')
   640→      const existing = await db.medicine.findUnique({ where: { id } })
   641→      if (!existing) return notFound('Medicine not found')
   642→      const parsed = await buildMedicineData(body.data, 'update')
   643→      if ('error' in parsed) return badRequest(parsed.error)
   644→      const newStock = typeof parsed.data.stock === 'number' ? parsed.data.stock : undefined
   645→      const medicine = await db.medicine.update({
   646→        where: { id },
   647→        data: parsed.data,
   648→        include: { category: true, extraImages: { orderBy: { sort: 'asc' } } },
   649→      })
   650→      if (newStock !== undefined && newStock !== existing.stock) {
   651→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: newStock - existing.stock, reason: 'MANUAL_EDIT', note: 'Manual stock update', userId: user.id }])
   652→      }
   653→      return Response.json({ medicine: medicineStaffJson(medicine) })
   654→    }
   655→
   656→    if (action === 'answer' || action === 'reject') {
   657→      const id = optStr(body.id)
   658→      if (!id) return badRequest('Question id is required')
   659→      const existing = await db.question.findUnique({ where: { id } })
   660→      if (!existing) return notFound('Question not found')
   661→
   662→      if (action === 'answer') {
   663→        const answer = typeof body.answer === 'string' ? body.answer.trim() : ''
   664→        if (answer.length < 2) return badRequest('Answer must be at least 2 characters')
   665→        if (answer.length > MAX_ANSWER_LENGTH) return badRequest(`Answer must be ${MAX_ANSWER_LENGTH} characters or less`)
   666→        await db.question.update({
   667→          where: { id },
   668→          data: { answer, status: 'ANSWERED', answeredById: user.id, answeredAt: new Date() },
   669→        })
   670→        // Notify the asker so they come back to read the answer (skip on re-answer of same content)
   671→        const medicineName = existing.medicineId
   672→          ? (await db.medicine.findUnique({ where: { id: existing.medicineId }, select: { name: true } }))?.name ?? 'a medicine'
   673→          : 'a medicine'
   674→        await notify(
   675→          existing.userId,
   676→          'Your question was answered',
   677→          `A pharmacist answered your question about ${medicineName}: "${answer.slice(0, 120)}${answer.length > 120 ? '…' : ''}"`
   678→        )
   679→      } else {
   680→        await db.question.update({
   681→          where: { id },
   682→          data: { status: 'REJECTED', answer: null, answeredById: null, answeredAt: null },
   683→        })
   684→      }
   685→
   686→      const updated = await db.question.findUnique({ where: { id }, include: questionInclude })
   687→      if (!updated) return notFound('Question not found')
   688→      return Response.json({ question: questionToJson(updated) })
   689→    }
   690→
   691→    return badRequest('Invalid action')
   692→  } catch (e) {
   693→    return serverError(e)
   694→  }
   695→}
