    60→/** Compact relative time for Q&A meta lines ("3 days ago") */
    61→function timeAgo(iso: string): string {
    62→  const s = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 1000))
    63→  if (s < 60) return 'just now'
    64→  const m = Math.floor(s / 60)
    65→  if (m < 60) return `${m} minute${m === 1 ? '' : 's'} ago`
    66→  const h = Math.floor(m / 60)
    67→  if (h < 24) return `${h} hour${h === 1 ? '' : 's'} ago`
    68→  const d = Math.floor(h / 24)
    69→  if (d < 30) return `${d} day${d === 1 ? '' : 's'} ago`
    70→  const mo = Math.floor(d / 30)
    71→  if (mo < 12) return `${mo} month${mo === 1 ? '' : 's'} ago`
    72→  const y = Math.floor(mo / 12)
    73→  return `${y} year${y === 1 ? '' : 's'} ago`
    74→}

   154→/**
   155→ * Product image gallery (Round 11). Progressive enhancement: the store's single
   156→ * image renders exactly as before until the detail fetch returns `images` —
   157→ * only then do the carousel controls + thumbnail strip appear. Fetch failures
   158→ * fall back silently to the store image (no error toast).

   160→function MedicineGallery({ medicine: m }: { medicine: Medicine }) {
   161→  const [gallery, setGallery] = useState<string[] | null>(null)
   162→  const [galleryLoading, setGalleryLoading] = useState(true)
   163→  const [index, setIndex] = useState(0)
   164→  const [dir, setDir] = useState(1)
   165→  const reduceMotion = useReducedMotion()
   166→
   167→  const firstImage = m.image
   168→  const id = m.id
   169→
   170→  // One detail fetch per opened product; the store image stays visible meanwhile
   171→  useEffect(() => {
   172→    if (!id) return
   173→    const ac = new AbortController()
   174→    api<{ medicine: Medicine }>(`/api/medicines?id=${encodeURIComponent(id)}`, { signal: ac.signal })
   175→      .then((d) => {
   176→        if (ac.signal.aborted) return
   177→        const imgs = d.medicine?.images
   178→        setGallery(
   179→          imgs && imgs.length > 0
   180→            ? Array.from(new Set(imgs)).slice(0, 6)
   181→            : dedupeImages([d.medicine?.image ?? firstImage])
   182→        )
   183→      })
   184→      .catch(() => {
   185→        // Silent fallback — the gallery is progressive enhancement, never an error surface
   186→        if (!ac.signal.aborted) setGallery(dedupeImages([firstImage]))
   187→      })
   188→      .finally(() => {
   189→        if (!ac.signal.aborted) setGalleryLoading(false)
   190→      })
   191→    return () => ac.abort()
   192→  }, [id, firstImage])
   193→
   194→  const images = gallery ?? dedupeImages([firstImage])
   195→  const multi = !galleryLoading && images.length > 1
   196→
   197→  const goto = (next: number, direction: 1 | -1) => {
   198→    if (images.length === 0) return
   199→    setDir(direction)
   200→    setIndex(((next % images.length) + images.length) % images.length) // wrap-around
   201→  }
   202→
   203→  // Single image (or still loading) — render exactly as before, zero regression
   204→  if (!multi) {
   205→    return <MedImage src={firstImage} alt={m.name} className="aspect-square w-full rounded-xl border bg-card" />
   206→  }
   207→
   208→  return (
   209→    <div>
   210→      <div
   211→        role="group"
   212→        aria-roledescription="carousel"
   213→        aria-label={`${m.name} images`}
   214→        tabIndex={0}
   215→        onKeyDown={(e) => {
   216→          if (e.key === 'ArrowLeft') {
   217→            e.preventDefault()
   218→            goto(index - 1, -1)
   219→          } else if (e.key === 'ArrowRight') {
   220→            e.preventDefault()
   221→            goto(index + 1, 1)
   222→          }
   223→        }}
   224→        className="relative aspect-square w-full overflow-hidden rounded-xl border bg-card outline-none focus-visible:ring-2 focus-visible:ring-primary/30 focus-visible:ring-offset-2"
   225→      >
   226→        <AnimatePresence mode="popLayout" initial={false}>
   227→          <motion.div
   228→            key={index}
   229→            initial={{ opacity: 0, x: reduceMotion ? 0 : 16 * dir }}
   230→            animate={{ opacity: 1, x: 0 }}
   231→            exit={{ opacity: 0, x: reduceMotion ? 0 : -16 * dir }}
   232→            transition={{ duration: 0.18, ease: 'easeOut' }}
   233→            className="absolute inset-0"
   234→          >
   235→            <MedImage
   236→              src={images[index]}
   237→              alt={`${m.name} — image ${index + 1} of ${images.length}`}
   238→              className="h-full w-full"
   239→            />
   240→          </motion.div>
   241→        </AnimatePresence>
   242→
   243→        {/* Prev / next — wrap-around, never disabled */}
   244→        <button
   245→          type="button"
   246→          aria-label="Previous image"
   247→          onClick={() => goto(index - 1, -1)}
   248→          className="absolute left-2 top-1/2 z-10 flex size-8 -translate-y-1/2 items-center justify-center rounded-full border bg-background/85 shadow-sm backdrop-blur transition-colors hover:bg-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary/30"
   249→        >
   250→          <ChevronLeft className="size-4" aria-hidden="true" />
   251→        </button>
   252→        <button
   253→          type="button"
   254→          aria-label="Next image"
   255→          onClick={() => goto(index + 1, 1)}
   256→          className="absolute right-2 top-1/2 z-10 flex size-8 -translate-y-1/2 items-center justify-center rounded-full border bg-background/85 shadow-sm backdrop-blur transition-colors hover:bg-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary/30"
   257→        >
   258→          <ChevronRight className="size-4" aria-hidden="true" />
   259→        </button>
   260→
   261→        {/* Position counter chip */}
   262→        <span
   263→          aria-live="polite"
   264→          aria-label={`Image ${index + 1} of ${images.length}`}
   265→          className="absolute bottom-2 right-2 z-10 rounded-full border bg-background/85 px-2 py-0.5 text-[11px] font-medium tabular-nums"
   266→        >
   267→          {index + 1}/{images.length}
   268→        </span>
   269→      </div>
   270→
   271→      {/* Thumbnail strip — active thumb gets the primary ring */}
   272→      <div className="mt-2 flex gap-2 overflow-x-auto pb-1 scrollbar-thin">
   273→        {images.map((src, i) => (
   274→          <button
   275→            key={`${i}-${src}`}
   276→            type="button"
   277→            aria-label={`Show image ${i + 1}`}
   278→            aria-current={i === index}
   279→            onClick={() => {
   280→              setDir(i > index ? 1 : -1)
   281→              setIndex(i)
   282→            }}
   283→            className={cn(
   284→              'size-14 shrink-0 overflow-hidden rounded-lg border transition-opacity focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-primary/30',
   285→              i === index ? 'border-primary ring-2 ring-primary/30' : 'opacity-70 hover:opacity-100'
   286→            )}
   287→          >
   288→            <MedImage src={src} alt="" className="h-full w-full" />
   289→          </button>
   290→        ))}
   291→      </div>
   292→    </div>
   293→  )
   294→}

   296→/** Inner body — keyed by medicine id so qty/adding state resets per product */
   297→function DetailBody({ medicine: m }: { medicine: Medicine }) {
   298→  const setDetailMedicine = useAppStore((s) => s.setDetailMedicine)
   299→  const [qty, setQty] = useState(1)
   300→  const [adding, setAdding] = useState(false)
   301→
   302→  const price = effectivePrice(m)
   303→  const stock = stockLabel(m.stock)
   304→  const out = m.stock <= 0
   305→  const maxQty = Math.max(1, m.stock)
   306→
   307→  // Freshness chip: reassure customers about shelf life (emerald ≥ 6 months, amber 3–6, red < 3)
   308→  let expiryChip: { label: string; className: string; title: string } | null = null
   309→  if (m.expiryDate) {
   310→    const days = Math.ceil((new Date(m.expiryDate).getTime() - Date.now()) / 86_400_000)
   311→    const months = Math.max(0, Math.round(days / 30))
   312→    if (days < 0) {
   313→      expiryChip = { label: 'Expired', className: 'bg-red-100 text-red-700', title: 'This batch has expired' }
   314→    } else if (days < 90) {
   315→      expiryChip = { label: 'Expires soon', className: 'bg-red-100 text-red-700', title: 'Less than 3 months shelf life left' }
   316→    } else if (days < 180) {
   317→      expiryChip = { label: `${months} months left`, className: 'bg-amber-100 text-amber-700', title: 'Short-dated batch — discounted freshness' }
   318→    } else {
   319→      expiryChip = { label: `${months} months left`, className: 'bg-emerald-100 text-emerald-700', title: 'Plenty of shelf life' }
   320→    }
   321→  }
   322→
   323→  const changeQty = (delta: number) => {
   324→    setQty((q) => Math.min(Math.max(1, q + delta), maxQty))
   325→  }
   326→
   327→  const handleAdd = async () => {
   328→    setAdding(true)
   329→    const ok = await addToCart(m, qty)
   330→    setAdding(false)
   331→    if (ok) setDetailMedicine(null)
   332→  }
   333→
   334→  return (
   335→    <div className="grid gap-6 p-4 sm:p-6 md:grid-cols-2">
   336→      {/* Image (gallery-aware — single image renders unchanged) */}
   337→      <div className="relative">
   338→        <MedicineGallery medicine={m} />
   339→        {m.requiresPrescription && (
   340→          <span className="absolute left-3 top-3 inline-flex items-center gap-1 rounded-md bg-amber-400 px-2 py-1 text-xs font-semibold text-amber-950 shadow-sm">
   341→            <FileWarning className="size-3.5" aria-hidden="true" />
   342→            Prescription required
   343→          </span>
   344→        )}
   345→      </div>
   346→
   347→      {/* Info */}
   348→      <div className="flex min-w-0 flex-col">
   349→        <h2 className="text-xl font-bold leading-tight tracking-tight">{m.name}</h2>
   350→        <p className="mt-1 text-sm text-muted-foreground">
   351→          {[m.brand, m.manufacturer].filter(Boolean).join(' · ') || '—'}
   352→        </p>
   353→        {m.genericName && (
   354→          <p className="mt-0.5 text-sm text-muted-foreground">
   355→            Generic: <span className="font-medium text-foreground">{m.genericName}</span>
   356→          </p>
   357→        )}
   358→
   359→        <p className="mt-3 text-2xl font-extrabold tabular-nums text-primary">{fmtBDT(price)}</p>
   360→        {price < m.price && (
   361→          <p className="text-sm tabular-nums text-muted-foreground">
   362→            <span className="line-through">{fmtBDT(m.price)}</span>{' '}
   363→            <span className="font-medium text-red-600">save {fmtBDT(m.price - price)}</span>
   364→          </p>
   365→        )}
   366→
   367→        {m.description && (
   368→          <p className="mt-3 line-clamp-4 text-sm leading-relaxed text-muted-foreground">
   369→            {m.description}
   370→          </p>
   371→        )}
   372→
   373→        <Separator className="my-4" />
   374→
   375→        <div className="space-y-2 text-sm">
   376→          <div className="flex items-center justify-between gap-3">
   377→            <span className="flex items-center gap-2 text-muted-foreground">
   378→              <Package className="size-4" aria-hidden="true" />
   379→              Stock
   380→            </span>
   381→            <span className={cn('font-semibold', stockToneClass[stock.tone])}>{stock.text}</span>
   382→          </div>
   383→          <div className="flex items-center justify-between gap-3">
   384→            <span className="flex items-center gap-2 text-muted-foreground">
   385→              <Tags className="size-4" aria-hidden="true" />
   386→              Unit
   387→            </span>
   388→            <span className="font-medium">{m.unit}</span>
   389→          </div>
   390→          <div className="flex items-center justify-between gap-3">
   391→            <span className="flex items-center gap-2 text-muted-foreground">
   392→              <CalendarClock className="size-4" aria-hidden="true" />
   393→              Expiry
   394→            </span>
   395→            <span className="flex items-center gap-1.5">
   396→              <span className="font-medium">{fmtDate(m.expiryDate)}</span>

   398→                <span

   405→            </span>

   409→        {m.requiresPrescription && (
   410→          <Alert className="mt-4 border-amber-300 bg-amber-50 text-amber-900">
   411→            <FileWarning className="size-4 text-amber-600" aria-hidden="true" />

   413→            <AlertDescription className="text-amber-800">

   421→          <div>
   422→            <p className="text-xs text-muted-foreground">Quantity</p>

   424→              <Button
   425→                variant="outline"
   426→                size="icon"

   435→                className="w-10 text-center text-lg font-semibold"

   444→                className="size-11 rounded-xl"

   447→                aria-label="Increase quantity"
   448→              >
   449→                <Plus className="size-4" aria-hidden="true" />
   450→              </Button>
   451→            </div>
   452→          </div>
   453→          <div className="text-right">
   454→            <p className="text-xs text-muted-foreground">Subtotal</p>
   455→            <p className="text-lg font-bold tabular-nums">{fmtBDT(price * (out ? 0 : qty))}</p>
   456→          </div>
   457→        </div>
   458→
   459→        <Button
   460→          className="mt-5 h-11 w-full rounded-xl"
   461→          disabled={out || adding}
   462→          onClick={() => void handleAdd()}
   463→        >
   464→          <ShoppingCart className="size-4" aria-hidden="true" />
   465→          {out ? 'Out of Stock' : adding ? 'Adding…' : 'Add to cart'}
   466→        </Button>
   467→      </div>
   468→    </div>
   469→  )
   470→}

   476→function QASection({ medicineId }: { medicineId: string }) {
   477→  const user = useAppStore((s) => s.user)
   478→  const setAuthOpen = useAppStore((s) => s.setAuthOpen)
   479→  const [questions, setQuestions] = useState<MedicineQA[] | null>(null)
   480→  const [loading, setLoading] = useState(true)
   481→  const [failed, setFailed] = useState(false)
   482→  const [composerOpen, setComposerOpen] = useState(false)
   483→  const [draft, setDraft] = useState('')
   484→  const [submitting, setSubmitting] = useState(false)
   485→  const [reloadKey, setReloadKey] = useState(0)
   486→  // Per-question in-flight id — only the clicked Helpful button spins/disables
   487→  const [votingId, setVotingId] = useState<string | null>(null)
   488→  // Round 11 — inline edit of an own PENDING question (15-min window)
   489→  const [editingId, setEditingId] = useState<string | null>(null)
   490→  const [editDraft, setEditDraft] = useState('')
   491→  const [savingEdit, setSavingEdit] = useState(false)
   492→  // Ticking clock for the countdown hints — only runs while an editable row is visible
   493→  const [now, setNow] = useState(() => Date.now())
   494→
   495→  // Auth OPTIONAL on this GET — sending the token also surfaces own PENDING rows
   496→  useEffect(() => {
   497→    const ac = new AbortController()
   498→    api<{ questions: MedicineQA[] }>(
   499→      `/api/questions?medicineId=${encodeURIComponent(medicineId)}`,
   500→      { signal: ac.signal }
   501→    )
   502→      .then((d) => {
   503→        if (ac.signal.aborted) return
   504→        setQuestions(d.questions)
   505→        setFailed(false)
   506→      })
   507→      .catch((err) => {
   508→        if (ac.signal.aborted) return
   509→        setFailed(err instanceof Error)
   510→      })
   511→      .finally(() => {
   512→        if (!ac.signal.aborted) setLoading(false)
   513→      })
   514→    return () => ac.abort()
   515→  }, [medicineId, user?.id, reloadKey])
   516→
   517→  const answered = (questions ?? []).filter((q) => q.status === 'ANSWERED')
   518→  const pendingMine = (questions ?? []).filter((q) => q.status === 'PENDING')
   519→  const draftLen = draft.trim().length
   520→  const canSubmit = draftLen >= MIN_QUESTION && draftLen <= MAX_QUESTION && !submitting
   521→  const hasEditable = pendingMine.some((q) => q.canEdit)
   522→  const editLen = editDraft.trim().length
   523→  const canSaveEdit = editLen >= MIN_QUESTION && editLen <= MAX_QUESTION && !savingEdit
   524→
   525→  // Refresh 'Editable for Xm' hints every 30s — only while an editable row exists
   526→  useEffect(() => {
   527→    if (!hasEditable) return
   528→    const t = setInterval(() => setNow(Date.now()), 30_000)
   529→    return () => clearInterval(t)
   530→  }, [hasEditable])
   531→
   532→  /** Minutes left in the 15-min edit window (0 when passed — hint hides, server refetch clears canEdit) */
   533→  const editMinutesLeft = (createdAt: string) =>
   534→    Math.ceil((new Date(createdAt).getTime() + EDIT_WINDOW_MS - now) / 60_000)
   535→
   536→  const startEdit = (q: MedicineQA) => {
   537→    setEditingId(q.id) // only one row in edit mode at a time
   538→    setEditDraft(q.question)
   539→  }
   540→
   541→  const cancelEdit = () => {
   542→    setEditingId(null)
   543→    setEditDraft('')
   544→  }
   545→
   546→  const saveEdit = async (q: MedicineQA) => {
   547→    if (!canSaveEdit) return
   548→    setSavingEdit(true)
   549→    try {
   550→      const d = await api<{ question: MedicineQA }>('/api/questions', {
   551→        method: 'PUT',
   552→        body: { id: q.id, question: editDraft.trim() },
   553→      })
   554→      setQuestions((list) =>
   555→        (list ?? []).map((row) =>
   556→          row.id === q.id
   557→            ? { ...row, question: d.question.question, canEdit: d.question.canEdit }
   558→            : row
   559→        )
   560→      )
   561→      setEditingId(null)
   562→      setEditDraft('')
   563→      toast.success('Question updated')
   564→    } catch (err) {
   565→      const message = err instanceof Error ? err.message : 'Failed to update question'
   566→      if (/edit window has passed/i.test(message)) {
   567→        toast.error('The edit window has passed')
   568→        setEditingId(null)
   569→        setEditDraft('')
   570→        setReloadKey((k) => k + 1) // refetch — server recomputes canEdit
   571→      } else if (/only pending questions/i.test(message)) {
   572→        toast.error('Only pending questions can be edited')
   573→        setEditingId(null)
   574→        setEditDraft('')
   575→        setReloadKey((k) => k + 1)
   576→      } else {
   577→        toast.error(message) // keep the editor open so the draft isn't lost
   578→      }
   579→    } finally {
   580→      setSavingEdit(false)
   581→    }
   582→  }
   583→
   584→  const submitQuestion = async () => {
   585→    if (!canSubmit) return
   586→    setSubmitting(true)
   587→    try {
   588→      const d = await api<{ question: MedicineQA }>('/api/questions', {
   589→        method: 'POST',
   590→        body: { medicineId, question: draft.trim() },
   591→      })
   592→      setQuestions((prev) => [d.question, ...(prev ?? [])])
   593→      setComposerOpen(false)
   594→      setDraft('')
   595→      toast.success('Question submitted — our pharmacist will answer soon')
   596→    } catch (err) {
   597→      toast.error(err instanceof Error ? err.message : 'Failed to submit question')
   598→    } finally {
   599→      setSubmitting(false)
   600→    }
   601→  }
   602→
   603→  const onAskClick = () => {
   604→    if (!user) {
   605→      setAuthOpen(true)
   606→      return
   607→    }
   608→    setComposerOpen((o) => !o)
   609→  }
   610→
   611→  /** Optimistic helpful-vote toggle; guests get the AuthModal, never a request */
   612→  const toggleHelpful = async (q: MedicineQA) => {
   613→    if (!user) {
   614→      setAuthOpen(true)
   615→      return
   616→    }
   617→    if (votingId) return // one vote request at a time keeps rollback predictable
   618→    const snapshot = questions
   619→    const nextVoted = !(q.hasVoted ?? false)
   620→    setQuestions((list) =>
   621→      (list ?? []).map((row) =>
   622→        row.id === q.id
   623→          ? {
   624→              ...row,
   625→              hasVoted: nextVoted,
   626→              helpfulCount: Math.max(0, (row.helpfulCount ?? 0) + (nextVoted ? 1 : -1)),
   627→            }
   628→          : row
   629→      )
   630→    )
   631→    setVotingId(q.id)
   632→    try {
   633→      const d = await api<{ questionId: string; helpfulCount: number; voted: boolean }>(
   634→        '/api/questions/helpful',
   635→        { method: 'POST', body: { questionId: q.id } }
   636→      )
   637→      // Reconcile with server truth
   638→      setQuestions((list) =>
   639→        (list ?? []).map((row) =>
   640→          row.id === d.questionId ? { ...row, helpfulCount: d.helpfulCount, hasVoted: d.voted } : row
   641→        )
   642→      )
   643→    } catch (err) {
   644→      const message = err instanceof Error ? err.message : 'Failed to record your vote'
   645→      if (/no longer accepting|only answered/i.test(message)) {
   646→        // 400 — question was likely reverted to pending; refetch for fresh state
   647→        toast.error('This question is no longer accepting votes')
   648→        setReloadKey((k) => k + 1)
   649→      } else {
   650→        toast.error(message)
   651→        setQuestions(snapshot) // rollback the optimistic toggle
   652→      }
   653→    } finally {
   654→      setVotingId(null)
   655→    }
   656→  }
   657→
   658→  return (
   659→    <section aria-label="Questions and answers" className="space-y-4">
   660→      <div className="flex flex-wrap items-center gap-2">
   661→        <h3 className="flex items-center gap-2 text-base font-bold tracking-tight">
   662→          <MessageCircleQuestion className="size-5 text-primary" aria-hidden="true" />
   663→          Questions &amp; answers
   664→          {answered.length > 0 && (
   665→            <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary">
   666→              {answered.length}
   667→            </span>
   668→          )}
   669→        </h3>
   670→        <Button
   671→          variant="outline"
   672→          size="sm"
   673→          className="ml-auto h-9 gap-1.5 rounded-lg"
   674→          onClick={onAskClick}
   675→        >
   676→          <MessageSquarePlus className="size-4" aria-hidden="true" />
   677→          Ask a question
   678→        </Button>
   679→      </div>
   680→
   681→      {loading ? (
   682→        /* 2 gray bubble rows while fetching (pulse + shimmer sweep) */
   683→        <div className="space-y-3" aria-hidden="true">
   684→          {[0, 1].map((i) => (
   685→            <div key={i} className="space-y-2">
   686→              <div className="flex items-start gap-2.5">
   687→                <Skeleton className="relative size-7 shrink-0 overflow-hidden rounded-lg">
   688→                  <Shimmer />
   689→                </Skeleton>
   690→                <Skeleton
   691→                  className={cn('relative flex-1 overflow-hidden rounded-xl', i === 0 ? 'h-9' : 'h-7')}
   692→                >
   693→                  <Shimmer />
   694→                </Skeleton>
   695→              </div>
   696→              <div className="flex items-start gap-2.5 pl-9">
   697→                <Skeleton className="relative size-7 shrink-0 overflow-hidden rounded-lg">
   698→                  <Shimmer />
   699→                </Skeleton>
   700→                <Skeleton className="relative h-12 flex-1 overflow-hidden rounded-xl">
   701→                  <Shimmer />
   702→                </Skeleton>
   703→              </div>
   704→            </div>
   705→          ))}
   706→        </div>
   707→      ) : failed ? (
   708→        <div className="flex flex-col items-center gap-2 rounded-xl border border-dashed p-6 text-center">
   709→          <p className="text-sm text-muted-foreground">Couldn&apos;t load questions.</p>
   710→          <Button
   711→            variant="outline"
   712→            size="sm"
   713→            className="h-9 rounded-lg"
   714→            onClick={() => {
   715→              setLoading(true)
   716→              setReloadKey((k) => k + 1)
   717→            }}
   718→          >
   719→            Try again
   720→          </Button>
   721→        </div>
   722→      ) : (
   723→        <>
   724→          {/* Own pending questions */}
   725→          {pendingMine.length > 0 && (
   726→            <div className="space-y-2">
   727→              <p className="flex items-center gap-1.5 text-xs font-semibold text-amber-700 dark:text-amber-400">
   728→                <Clock3 className="size-3.5" aria-hidden="true" />
   729→                Your question (pending review)
   730→                <span className="rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-semibold text-amber-800 dark:bg-amber-950/60 dark:text-amber-300">
   731→                  Awaiting answer
   732→                </span>
   733→              </p>
   734→              {pendingMine.map((q, i) => {
   735→                const minutesLeft = editMinutesLeft(q.createdAt)
   736→                const isEditing = editingId === q.id
   737→                return (
   738→                  <motion.div
   739→                    key={q.id}
   740→                    initial={{ opacity: 0, y: 8 }}
   741→                    animate={{ opacity: 1, y: 0 }}
   742→                    transition={{ duration: 0.25, delay: Math.min(i * 0.04, 0.2), ease: 'easeOut' }}
   743→                    className="rounded-xl border border-dashed border-amber-200 bg-amber-50/50 p-3 dark:border-amber-900/70 dark:bg-amber-950/20"
   744→                  >
   745→                    {isEditing ? (
   746→                      /* Inline edit composer — same visual language as the Ask composer */
   747→                      <div>
   748→                        <Textarea
   749→                          value={editDraft}
   750→                          maxLength={MAX_QUESTION}
   751→                          rows={3}
   752→                          autoFocus
   753→                          aria-label="Edit your question"
   754→                          onChange={(e) => setEditDraft(e.target.value)}
   755→                          onKeyDown={(e) => {
   756→                            if (e.key === 'Escape') {
   757→                              // Cancel the editor, not the dialog
   758→                              e.stopPropagation()
   759→                              e.preventDefault()
   760→                              cancelEdit()
   761→                            }
   762→                          }}
   763→                          className="min-h-20 resize-none border-border bg-card focus-visible:ring-primary/30"
   764→                        />
   765→                        {editLen > 0 && editLen < MIN_QUESTION && (
   766→                          <p className="mt-1 text-[11px] text-amber-600">
   767→                            Question must be at least {MIN_QUESTION} characters.
   768→                          </p>
   769→                        )}
   770→                        <div className="mt-2 flex items-center justify-between gap-2">
   771→                          <span
   772→                            className={cn(
   773→                              'text-[11px] tabular-nums',
   774→                              editLen > MAX_QUESTION ? 'font-medium text-red-600' : 'text-muted-foreground'
   775→                            )}
   776→                            aria-live="polite"
   777→                          >
   778→                            {editDraft.length} / {MAX_QUESTION}
   779→                          </span>
   780→                          <div className="flex gap-2">
   781→                            <Button
   782→                              variant="ghost"
   783→                              size="sm"
   784→                              className="h-8 rounded-lg"
   785→                              disabled={savingEdit}
   786→                              onClick={cancelEdit}
   787→                            >
   788→                              Cancel
   789→                            </Button>
   790→                            <Button
   791→                              size="sm"
   792→                              className="h-8 gap-1.5 rounded-lg"
   793→                              disabled={!canSaveEdit}
   794→                              onClick={() => void saveEdit(q)}
   795→                            >
   796→                              {savingEdit && <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />}
   797→                              {savingEdit ? 'Saving…' : 'Save'}
   798→                            </Button>
   799→                          </div>
   800→                        </div>
   801→                      </div>
   802→                    ) : (
   803→                      <>
   804→                        <p className="text-sm leading-snug text-muted-foreground">{q.question}</p>
   805→                        <div className="mt-1.5 flex flex-wrap items-center gap-x-2 gap-y-1">
   806→                          <p className="text-[11px] text-muted-foreground">
   807→                            {q.askedByName} · {timeAgo(q.createdAt)}
   808→                          </p>
   809→                          {q.canEdit && (
   810→                            <>
   811→                              {minutesLeft > 0 && (
   812→                                <span
   813→                                  className="text-[11px] tabular-nums text-muted-foreground"
   814→                                  title="You can edit this question while it awaits review"
   815→                                >
   816→                                  Editable for {minutesLeft}m
   817→                                </span>
   818→                              )}
   819→                              <Button
   820→                                variant="ghost"
   821→                                size="sm"
   822→                                className="ml-auto h-7 gap-1 rounded-lg px-2 text-xs"
   823→                                title="You can edit this question for 15 minutes"
   824→                                aria-label={`Edit your question: ${q.question}`}
   825→                                onClick={() => startEdit(q)}
   826→                              >
   827→                                <Pencil className="size-3.5" aria-hidden="true" />
   828→                                Edit
   829→                              </Button>
   830→                            </>
   831→                          )}
   832→                        </div>
   833→                      </>
   834→                    )}
   835→                  </motion.div>
   836→                )
   837→              })}
   838→            </div>
   839→          )}
   840→
   841→          {/* Answered threads */}
   842→          {answered.length > 0 ? (
   843→            <div
   844→              className={cn(
   845→                'space-y-4',
   846→                answered.length > 4 && 'max-h-80 overflow-y-auto pr-1 scrollbar-thin'
   847→              )}
   848→            >
   849→              {answered.map((q, i) => (
   850→                <motion.div
   851→                  key={q.id}
   852→                  initial={{ opacity: 0, y: 8 }}
   853→                  animate={{ opacity: 1, y: 0 }}
   854→                  transition={{ duration: 0.25, delay: Math.min(i * 0.05, 0.3), ease: 'easeOut' }}
   855→                  className="space-y-2"
   856→                >
   857→                  {/* Question row */}
   858→                  <div className="flex items-start gap-2.5">
   859→                    <span
   860→                      className="flex size-7 shrink-0 items-center justify-center rounded-lg bg-amber-500/15 text-xs font-bold text-amber-700 dark:text-amber-400"
   861→                      aria-hidden="true"
   862→                    >
   863→                      Q
   864→                    </span>
   865→                    <div className="min-w-0 flex-1">
   866→                      <p className="text-sm font-medium leading-snug">{q.question}</p>
   867→                      <p className="mt-0.5 text-[11px] text-muted-foreground">
   868→                        {q.askedByName} · {timeAgo(q.createdAt)}
   869→                      </p>
   870→                    </div>
   871→                  </div>
   872→                  {/* Answer row — indented, pharmacist chip, helpful votes */}
   873→                  {q.answer && (
   874→                    <div className="flex items-start gap-2.5 pl-9">
   875→                      <span
   876→                        className="flex size-7 shrink-0 items-center justify-center rounded-lg bg-primary/15 text-xs font-bold text-primary"
   877→                        aria-hidden="true"
   878→                      >
   879→                        A
   880→                      </span>
   881→                      <div
   882→                        role="group"
   883→                        aria-label={
   884→                          `Answer from ${q.answerByName ?? 'Pharmacist'}` +
   885→                          ((q.helpfulCount ?? 0) > 0
   886→                            ? `, marked helpful by ${q.helpfulCount} ${q.helpfulCount === 1 ? 'person' : 'people'}`
   887→                            : '')
   888→                        }
   889→                        className="min-w-0 flex-1 rounded-xl rounded-tl-sm border bg-card/60 p-3"
   890→                      >
   891→                        <p className="inline-flex items-center gap-1 text-[11px] font-semibold text-primary">
   892→                          <BadgeCheck className="size-3.5" aria-hidden="true" />
   893→                          {q.answerByName ?? 'Pharmacist'}
   894→                        </p>
   895→                        <p className="mt-1 text-sm leading-relaxed text-muted-foreground">{q.answer}</p>
   896→                        <HelpfulButton
   897→                          question={q}
   898→                          busy={votingId === q.id}
   899→                          onToggle={() => void toggleHelpful(q)}
   900→                        />
   901→                      </div>
   902→                    </div>
   903→                  )}
   904→                </motion.div>
   905→              ))}
   906→            </div>
   907→          ) : (
   908→            pendingMine.length === 0 && (
   909→              <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed p-6 text-center">
   910→                <span className="flex size-11 items-center justify-center rounded-full bg-primary/10">
   911→                  <MessageCircleQuestion className="size-5 text-primary" aria-hidden="true" />
   912→                </span>
   913→                <p className="text-sm font-semibold">No questions yet</p>
   914→                <p className="text-xs text-muted-foreground">
   915→                  Be the first to ask about this medicine.
   916→                </p>
   917→              </div>
   918→            )
   919→          )}
   920→
   921→          {/* Inline composer (signed-in only — guests get the auth modal) */}
   922→          {composerOpen && user && (
   923→            <motion.div
   924→              initial={{ opacity: 0, y: 6 }}
   925→              animate={{ opacity: 1, y: 0 }}
   926→              transition={{ duration: 0.2, ease: 'easeOut' }}
   927→              className="rounded-xl border bg-card/50 p-3"
   928→            >
   929→              <Textarea
   930→                value={draft}
   931→                maxLength={MAX_QUESTION}
   932→                onChange={(e) => setDraft(e.target.value)}
   933→                placeholder="Ask about dosage, usage, interactions…"
   934→                aria-label="Your question"
   935→                className="min-h-20 resize-none focus-visible:ring-primary/30"
   936→              />
   937→              {draftLen > 0 && draftLen < MIN_QUESTION && (
   938→                <p className="mt-1 text-[11px] text-amber-600">
   939→                  Question must be at least {MIN_QUESTION} characters.
   940→                </p>
   941→              )}
   942→              <div className="mt-2 flex items-center justify-between gap-2">
   943→                <span
   944→                  className={cn(
   945→                    'text-[11px] tabular-nums',
   946→                    draftLen > 540 ? 'font-medium text-amber-600' : 'text-muted-foreground'
   947→                  )}
   948→                  aria-live="polite"
   949→                >
   950→                  {draft.length} / {MAX_QUESTION}
   951→                </span>
   952→                <div className="flex gap-2">
   953→                  <Button
   954→                    variant="ghost"
   955→                    size="sm"
   956→                    className="h-9 rounded-lg"
   957→                    disabled={submitting}
   958→                    onClick={() => {
   959→                      setComposerOpen(false)
   960→                      setDraft('')
   961→                    }}
   962→                  >
   963→                    Cancel
   964→                  </Button>
   965→                  <Button
   966→                    size="sm"
   967→                    className="h-9 gap-1.5 rounded-lg"
   968→                    disabled={!canSubmit}
   969→                    onClick={() => void submitQuestion()}

   971→                    {submitting && <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />}

   973→                  </Button>
   974→                </div>
   975→              </div>
   976→            </motion.div>
   977→          )}
   978→        </>
   979→      )}
   980→    </section>
   981→  )
