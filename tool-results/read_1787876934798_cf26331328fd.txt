    51→    51→function timeAgo(iso: string): string {
    52→    52→  const s = Math.max(0, Math.floor((Date.now() - new Date(iso).getTime()) / 1000))
    53→    53→  if (s < 60) return 'just now'
    54→    54→  const m = Math.floor(s / 60)
    55→    55→  if (m < 60) return `${m} minute${m === 1 ? '' : 's'} ago`
    56→    56→  const h = Math.floor(m / 60)
    57→    57→  if (h < 24) return `${h} hour${h === 1 ? '' : 's'} ago`
    58→    58→  const d = Math.floor(h / 24)
    59→    59→  if (d < 30) return `${d} day${d === 1 ? '' : 's'} ago`
    60→    60→  const mo = Math.floor(d / 30)
    61→    61→  if (mo < 12) return `${mo} month${mo === 1 ? '' : 's'} ago`
    62→    62→  const y = Math.floor(mo / 12)
    63→    63→  return `${y} year${y === 1 ? '' : 's'} ago`
    64→    64→}

   324→   324→/**
   325→   325→ * Product Q&A — public answered threads (chat bubbles) + the viewer's own
   326→   326→ * pending questions. Guests are prompted to sign in before asking.
   327→   327→ */
   328→   328→function QASection({ medicineId }: { medicineId: string }) {
   329→   329→  const user = useAppStore((s) => s.user)
   330→   330→  const setAuthOpen = useAppStore((s) => s.setAuthOpen)
   331→   331→  const [questions, setQuestions] = useState<MedicineQA[] | null>(null)
   332→   332→  const [loading, setLoading] = useState(true)
   333→   333→  const [failed, setFailed] = useState(false)
   334→   334→  const [composerOpen, setComposerOpen] = useState(false)
   335→   335→  const [draft, setDraft] = useState('')
   336→   336→  const [submitting, setSubmitting] = useState(false)
   337→   337→  const [reloadKey, setReloadKey] = useState(0)
   338→   338→  // Per-question in-flight id — only the clicked Helpful button spins/disables
   339→   339→  const [votingId, setVotingId] = useState<string | null>(null)
   340→   340→
   341→   341→  // Auth OPTIONAL on this GET — sending the token also surfaces own PENDING rows
   342→   342→  useEffect(() => {
   343→   343→    const ac = new AbortController()
   344→   344→    api<{ questions: MedicineQA[] }>(
   345→   345→      `/api/questions?medicineId=${encodeURIComponent(medicineId)}`,
   346→   346→      { signal: ac.signal }
   347→   347→    )
   348→   348→      .then((d) => {
   349→   349→        if (ac.signal.aborted) return
   350→   350→        setQuestions(d.questions)
   351→   351→        setFailed(false)
   352→   352→      })
   353→   353→      .catch((err) => {
   354→   354→        if (ac.signal.aborted) return
   355→   355→        setFailed(err instanceof Error)
   356→   356→      })
   357→   357→      .finally(() => {
   358→   358→        if (!ac.signal.aborted) setLoading(false)
   359→   359→      })
   360→   360→    return () => ac.abort()
   361→   361→  }, [medicineId, user?.id, reloadKey])
   362→   362→
   363→   363→  const answered = (questions ?? []).filter((q) => q.status === 'ANSWERED')
   364→   364→  const pendingMine = (questions ?? []).filter((q) => q.status === 'PENDING')
   365→   365→  const draftLen = draft.trim().length
   366→   366→  const canSubmit = draftLen >= MIN_QUESTION && draftLen <= MAX_QUESTION && !submitting
   367→   367→
   368→   368→  const submitQuestion = async () => {
   369→   369→    if (!canSubmit) return
   370→   370→    setSubmitting(true)
   371→   371→    try {
   372→   372→      const d = await api<{ question: MedicineQA }>('/api/questions', {
   373→   373→        method: 'POST',
   374→   374→        body: { medicineId, question: draft.trim() },
   375→   375→      })
   376→   376→      setQuestions((prev) => [d.question, ...(prev ?? [])])
   377→   377→      setComposerOpen(false)
   378→   378→      setDraft('')
   379→   379→      toast.success('Question submitted — our pharmacist will answer soon')
   380→   380→    } catch (err) {
   381→   381→      toast.error(err instanceof Error ? err.message : 'Failed to submit question')
   382→   382→    } finally {
   383→   383→      setSubmitting(false)
   384→   384→    }
   385→   385→  }
   386→   386→
   387→   387→  const onAskClick = () => {
   388→   388→    if (!user) {
   389→   389→      setAuthOpen(true)
   390→   390→      return
   391→   391→    }
   392→   392→    setComposerOpen((o) => !o)
   393→   393→  }
   394→   394→
   395→   395→  /** Optimistic helpful-vote toggle; guests get the AuthModal, never a request */
   396→   396→  const toggleHelpful = async (q: MedicineQA) => {
   397→   397→    if (!user) {
   398→   398→      setAuthOpen(true)
   399→   399→      return
   400→   400→    }
   401→   401→    if (votingId) return // one vote request at a time keeps rollback predictable
   402→   402→    const snapshot = questions
   403→   403→    const nextVoted = !(q.hasVoted ?? false)
   404→   404→    setQuestions((list) =>
   405→   405→      (list ?? []).map((row) =>
   406→   406→        row.id === q.id
   407→   407→          ? {
   408→   408→              ...row,
   409→   409→              hasVoted: nextVoted,
   410→   410→              helpfulCount: Math.max(0, (row.helpfulCount ?? 0) + (nextVoted ? 1 : -1)),
   411→   411→            }
   412→   412→          : row
   413→   413→      )
   414→   414→    )
   415→   415→    setVotingId(q.id)
   416→   416→    try {
   417→   417→      const d = await api<{ questionId: string; helpfulCount: number; voted: boolean }>(
   418→   418→        '/api/questions/helpful',
   419→   419→        { method: 'POST', body: { questionId: q.id } }
   420→   420→      )
   421→   421→      // Reconcile with server truth
   422→   422→      setQuestions((list) =>
   423→   423→        (list ?? []).map((row) =>
   424→   424→          row.id === d.questionId ? { ...row, helpfulCount: d.helpfulCount, hasVoted: d.voted } : row
   425→   425→        )
   426→   426→      )
   427→   427→    } catch (err) {
   428→   428→      const message = err instanceof Error ? err.message : 'Failed to record your vote'
   429→   429→      if (/no longer accepting|only answered/i.test(message)) {
   430→   430→        // 400 — question was likely reverted to pending; refetch for fresh state
   431→   431→        toast.error('This question is no longer accepting votes')
   432→   432→        setReloadKey((k) => k + 1)
   433→   433→      } else {
   434→   434→        toast.error(message)
   435→   435→        setQuestions(snapshot) // rollback the optimistic toggle
   436→   436→      }
   437→   437→    } finally {
   438→   438→      setVotingId(null)
   439→   439→    }
   440→   440→  }
   441→   441→
   442→   442→  return (
   443→   443→    <section aria-label="Questions and answers" className="space-y-4">
   444→   444→      <div className="flex flex-wrap items-center gap-2">
   445→   445→        <h3 className="flex items-center gap-2 text-base font-bold tracking-tight">
   446→   446→          <MessageCircleQuestion className="size-5 text-primary" aria-hidden="true" />
   447→   447→          Questions &amp; answers
   448→   448→          {answered.length > 0 && (
   449→   449→            <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary">
   450→   450→              {answered.length}
   451→   451→            </span>

   453→   453→        </h3>
