    67→function shortDay(d: string): string {
    68→  const date = new Date(d)
    69→  if (Number.isNaN(date.getTime())) return d
    70→  return date.toLocaleDateString('en-GB', { day: 'numeric', month: 'short' })

   185→function DeliverySection({
   186→  staff,
   187→  loading,
   188→  error,

   190→  onRetry,

   201→    const list = staff ?? []
   202→    return {
   203→      delivered: list.reduce((sum, s) => sum + s.delivered, 0),
   204→      active: list.reduce((sum, s) => sum + s.active, 0),
   205→      cod: list.reduce((sum, s) => sum + s.codCollected, 0),
   206→      size: list.length,
   207→    }
   208→  }, [staff])
   209→
   210→  if (loading && !staff) return <DeliverySkeleton />
   211→
   212→  return (
   213→    <div className="space-y-4">
   214→      {/* Section header + toolbar */}
   215→      <div className="flex flex-wrap items-center justify-between gap-2">
   216→        <div>
   217→          <h2 className="flex items-center gap-2 text-lg font-semibold tracking-tight">
   218→            <Truck className="h-5 w-5 text-primary" aria-hidden="true" /> Delivery performance
   219→          </h2>
   220→          <p className="text-sm text-muted-foreground">How each delivery partner is doing</p>
   221→        </div>
   222→        <Button
   223→          variant="outline"
   224→          onClick={onExport}
   225→          disabled={exportPending || !staff || staff.length === 0}
   226→        >
   227→          {exportPending ? (
   228→            <Loader2 className="h-4 w-4 animate-spin" />
   229→          ) : (
   230→            <Download className="h-4 w-4" />
   231→          )}
   232→          Export CSV
   233→        </Button>
   234→      </div>
   235→
   236→      {error ? (
   237→        <Card className="flex-col items-center gap-2 p-10 text-center">
   238→          <Truck className="h-8 w-8 text-muted-foreground/50" />
   239→          <p className="text-sm text-muted-foreground">{error}</p>
   240→          <Button variant="outline" size="sm" onClick={onRetry}>
   241→            <RefreshCw className="h-3.5 w-3.5" /> Retry

   243→        </Card>

   247→          <div>
   248→            <p className="text-sm font-medium">No delivery staff yet</p>
   249→            <p className="text-sm text-muted-foreground">

   251→            </p>
   252→          </div>
   253→        </Card>

   255→        <>
   256→          {/* Team summary strip */}
   257→          <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   258→            <StatChip
   259→              label="Total delivered"
   260→              value={totals.delivered}
   261→              sub="orders completed by the team"
   262→              tone="bg-emerald-100"
   263→              icon={<PackageCheck className="size-4 text-emerald-700" aria-hidden="true" />}

   266→              label="Active on the road"
   267→              value={totals.active}

   271→            />
   272→            <StatChip
   273→              label="COD collected"
   274→              value={fmtBDT(totals.cod)}
   275→              sub="cash collected on delivery"
   276→              tone="bg-amber-100"
   277→              icon={<HandCoins className="size-4 text-amber-700" aria-hidden="true" />}
   278→            />
   279→            <StatChip
   280→              label="Team size"
   281→              value={totals.size}
   282→              sub="delivery partners"
   283→              tone="bg-primary/10"
   284→              icon={<Users className="text-primary size-4" aria-hidden="true" />}
   285→            />
   286→          </div>
   287→
   288→          {/* Staff table */}
   289→          <Card className="gap-0 py-4">
   290→            <CardContent className="px-4">
   291→              <div className="max-h-[60vh] overflow-auto rounded-md border scrollbar-thin">
   292→                <Table className="min-w-[820px]" aria-label="Delivery performance by staff member">
   293→                  <TableHeader className="sticky top-0 z-10 bg-card shadow-[0_1px_0_0_var(--border)]">
   294→                    <TableRow className="hover:bg-transparent">
   295→                      <TableHead>Staff</TableHead>
   296→                      <TableHead className="text-center">Assigned</TableHead>
   297→                      <TableHead className="text-center">Active</TableHead>
   298→                      <TableHead className="text-center">Delivered</TableHead>
   299→                      <TableHead className="text-center">Failed</TableHead>
   300→                      <TableHead className="text-center">Success rate</TableHead>
   301→                      <TableHead className="text-right">COD collected</TableHead>
   302→                      <TableHead className="text-right">Avg completion</TableHead>
   303→                      <TableHead className="text-right">Last delivery</TableHead>
   304→                    </TableRow>
   305→                  </TableHeader>
   306→                  <TableBody>
   307→                    {staff.map((s) => {
   308→                      const hasOrders = s.assigned > 0 || s.delivered > 0
   309→                      const successPct =
   310→                        s.assigned > 0 ? Math.round((s.delivered / s.assigned) * 100) : null
   311→                      const avgH =
   312→                        s.avgCompletionHours != null
   313→                          ? Math.round(s.avgCompletionHours * 10) / 10
   314→                          : null
   315→                      const initial = (s.name?.trim()?.[0] ?? s.email[0] ?? '?').toUpperCase()
   316→                      return (
   317→                        <TableRow
   318→                          key={s.id}
   319→                          className={cn(!hasOrders && 'text-muted-foreground')}
   320→                        >
   321→                          <TableCell>
   322→                            <div className="flex items-center gap-2.5">
   323→                              <Avatar className="size-8">
   324→                                <AvatarFallback className="text-xs">{initial}</AvatarFallback>
   325→                              </Avatar>
   326→                              <div className="min-w-0">
   327→                                <p className="truncate text-sm font-medium">{s.name ?? s.email}</p>
   328→                                <p className="max-w-44 truncate text-xs text-muted-foreground">
   329→                                  {s.email}
   330→                                </p>
   331→                              </div>
   332→                            </div>
   333→                          </TableCell>
   334→                          <TableCell className="whitespace-nowrap text-center text-sm">
   335→                            {s.assigned}
   336→                          </TableCell>
   337→                          <TableCell className="text-center">
   338→                            {s.active > 0 ? (
   339→                              <Badge
   340→                                variant="outline"
   341→                                className="whitespace-nowrap border-teal-300 bg-teal-100 text-teal-800"
   342→                              >
   343→                                {s.active}
   344→                              </Badge>
   345→                            ) : (
   346→                              <span className="text-sm">0</span>
   347→                            )}
   348→                          </TableCell>
   349→                          <TableCell
   350→                            className={cn(
   351→                              'whitespace-nowrap text-center text-sm',
   352→                              hasOrders && 'font-bold text-emerald-600 dark:text-emerald-400'
   353→                            )}
   354→                          >
   355→                            {s.delivered}
   356→                          </TableCell>
   357→                          <TableCell
   358→                            className={cn(
   359→                              'whitespace-nowrap text-center text-sm',
   360→                              s.failed > 0 && 'font-medium text-red-600 dark:text-red-400'
   361→                            )}
   362→                          >
   363→                            {s.failed}
   364→                          </TableCell>
   365→                          <TableCell>
   366→                            {successPct === null ? (
   367→                              <span className="text-sm text-muted-foreground">—</span>
   368→                            ) : (
   369→                              <div
   370→                                className="flex items-center justify-center gap-2"
   371→                                title={`${successPct}% of assigned orders delivered`}
   372→                              >
   373→                                <Progress
   374→                                  value={successPct}
   375→                                  className="h-1.5 w-16"
   376→                                  aria-label={`${successPct}% success rate`}
   377→                                />
   378→                                <span className="w-9 text-xs font-medium tabular-nums">
   379→                                  {successPct}%
   380→                                </span>
   381→                              </div>
   382→                            )}
   383→                          </TableCell>
   384→                          <TableCell className="whitespace-nowrap text-right text-sm font-medium">
   385→                            {fmtBDT(s.codCollected)}
   386→                          </TableCell>
   387→                          <TableCell
   388→                            className="whitespace-nowrap text-right text-sm"
   389→                            title="Avg hours from order placed to delivered"
   390→                          >
   391→                            {avgH != null ? `${avgH}h` : '—'}
   392→                          </TableCell>
   393→                          <TableCell className="whitespace-nowrap text-right text-sm">
   394→                            {s.lastDeliveryAt ? (
   395→                              fmtDateTime(s.lastDeliveryAt)
   396→                            ) : (
   397→                              <span className="text-muted-foreground">Never</span>
   398→                            )}
   399→                          </TableCell>
   400→                        </TableRow>
   401→                      )
   402→                    })}
   403→                  </TableBody>
   404→                </Table>
   405→              </div>
   406→            </CardContent>
   407→          </Card>
   408→        </>
   409→      )}
   410→    </div>
   411→  )
   412→}

   414→export default function AdminReports() {
   415→  const [days, setDays] = useState('7')
   416→  const [reports, setReports] = useState<ReportsData | null>(null)
   417→  const [loading, setLoading] = useState(true)
   418→  const [exportPending, setExportPending] = useState(false)
   419→  const [staff, setStaff] = useState<DeliveryStaff[] | null>(null)
   420→  const [staffLoading, setStaffLoading] = useState(true)
   421→  const [staffError, setStaffError] = useState<string | null>(null)
   422→  const [staffExportPending, setStaffExportPending] = useState(false)
   423→
   424→  const load = useCallback(async () => {
   425→    setLoading(true)
   426→    try {
   427→      const d = await api<ReportsData>(`/api/admin?resource=reports&days=${days}`)
   428→      setReports(d)
   429→    } catch (e) {
   430→      toast.error(e instanceof Error ? e.message : 'Failed to load reports')
   431→    } finally {
   432→      setLoading(false)
   433→    }
   434→  }, [days])
   435→
   436→  useEffect(() => {
   437→    void load()
   438→  }, [load])
   439→
   440→  const loadDelivery = useCallback(async (signal?: AbortSignal) => {
   441→    setStaffLoading(true)
   442→    setStaffError(null)
   443→    try {
   444→      const d = await api<{ staff: DeliveryStaff[] }>(
   445→        '/api/admin?resource=delivery-performance',
   446→        { signal }
   447→      )
   448→      setStaff(d.staff)
   449→    } catch (e) {
   450→      if (e instanceof DOMException && e.name === 'AbortError') return
   451→      setStaffError(e instanceof Error ? e.message : 'Failed to load delivery performance')
   452→    } finally {
   453→      setStaffLoading(false)
   454→    }
   455→  }, [])
   456→
   457→  useEffect(() => {
   458→    const ctrl = new AbortController()
   459→    void loadDelivery(ctrl.signal)
   460→    return () => ctrl.abort()
   461→  }, [loadDelivery])
   462→
   463→  async function exportCsv() {
   464→    setExportPending(true)
   465→    try {
   466→      const d = await api<{ filename: string; csv: string }>(
   467→        `/api/admin?resource=export-report&days=${days}`
   468→      )
   469→      if (typeof d.csv !== 'string' || typeof d.filename !== 'string')
   470→        throw new Error('Export unavailable')
   471→      downloadCsv(d.filename, d.csv)
   472→      toast.success('Report exported')
   473→    } catch (e) {
   474→      toast.error(e instanceof Error ? e.message : 'Failed to export report')
   475→    } finally {
   476→      setExportPending(false)
   477→    }
   478→  }
   479→
   480→  async function exportDeliveryCsv() {

   482→    try {
   483→      const d = await api<{ filename: string; csv: string }>(
   484→        '/api/admin?resource=export-delivery'
   485→      )
   486→      if (typeof d.csv !== 'string' || typeof d.filename !== 'string')
   487→        throw new Error('Export unavailable')
   488→      downloadCsv(d.filename, d.csv)
   489→      toast.success('Delivery report exported')
   490→    } catch (e) {
   491→      toast.error(e instanceof Error ? e.message : 'Failed to export delivery report')
   492→    } finally {
   493→      setStaffExportPending(false)
   494→    }
   495→  }
   496→
   497→  const catHeight = reports ? Math.max(220, reports.categorySales.length * 44 + 40) : 220

   499→  return (
   500→    <div className="space-y-4 md:space-y-6">

   504→          <SelectTrigger className="w-[160px]">

   507→          <SelectContent>
   508→            <SelectItem value="7">Last 7 days</SelectItem>
   509→            <SelectItem value="30">Last 30 days</SelectItem>
   510→          </SelectContent>
   511→        </Select>
   512→        <Button variant="outline" onClick={() => void exportCsv()} disabled={exportPending}>
   513→          {exportPending ? (
   514→            <Loader2 className="h-4 w-4 animate-spin" />
   515→          ) : (
   516→            <Download className="h-4 w-4" />
   517→          )}
   518→          Export CSV
   519→        </Button>
   520→      </div>
   521→
   522→      {loading && !reports ? (
   523→        <ReportsSkeleton />
   524→      ) : !reports ? (
   525→        <Card className="items-center p-10 text-center">
   526→          <BarChart3 className="h-8 w-8 text-muted-foreground/50" />
   527→          <p className="text-sm text-muted-foreground">Could not load reports.</p>

   530→        <>
   531→          {/* Charts */}

   533→            <Card>
   534→              <CardHeader>
   535→                <CardTitle>Sales by day</CardTitle>
   536→                <CardDescription>Revenue and order volume</CardDescription>
   537→              </CardHeader>
   538→              <CardContent>
   539→                <ResponsiveContainer width="100%" height={300}>
   540→                  <BarChart data={reports.salesByDay} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   541→                    <CartesianGrid strokeDasharray="3 3" stroke={CHART_GRID} vertical={false} />
   542→                    <XAxis

   544→                      tickFormatter={shortDay}

   548→                    <YAxis

   550→                      orientation="left"
   551→                      tick={{ fontSize: 11 }}
   552→                      stroke={AXIS}

   554→                      tickFormatter={kFmt}

   558→                      orientation="right"
   559→                      tick={{ fontSize: 11 }}
   560→                      stroke={AXIS}
   561→                      width={32}

   564→                    <Tooltip

   566→                      formatter={(v, name) => (name === 'orders' ? `${Number(v)} orders` : fmtBDT(Number(v)))}
   567→                      labelFormatter={(l) => fmtDate(String(l))}
   568→                    />
   569→                    <Legend
   570→                      wrapperStyle={{ fontSize: 12 }}
   571→                      formatter={(v) => (v === 'orders' ? 'Orders' : 'Revenue')}
   572→                    />
   573→                    <Bar
   574→                      yAxisId="rev"
   575→                      dataKey="revenue"
   576→                      fill="#10b981"
   577→                      radius={[4, 4, 0, 0]}

   579→                    />
