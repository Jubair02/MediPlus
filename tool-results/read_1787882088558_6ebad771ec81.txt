    96→    96→function normalizeImageSrc(src: string): string {
    97→    97→  const s = src.trim()
    98→    98→  if (s.startsWith('/') && typeof window !== 'undefined') return `${window.location.origin}${s}`
    99→    99→  return s
   100→   100→}

   102→   102→function isValidImageSrc(src: string): boolean {
   103→   103→  const s = normalizeImageSrc(src)
   104→   104→  return s.startsWith('data:image') || /^https?:\/\//i.test(s)
   105→   105→}

   129→   129→  function addUrl() {
   130→   130→    const src = urlInput.trim()

   133→   133→      toast.error('Image must be an http(s) URL or an image data URL')
   134→   134→      return

   140→   140→  async function onFilesPicked(files: FileList | null) {
   141→   141→    if (!files || files.length === 0 || processing) {
   142→   142→      resetFileInput()

   144→   144→    }
   145→   145→    if (atCap) {

   147→   147→      resetFileInput()

   150→   150→    const picked = Array.from(files)
   151→   151→    const images = picked.filter((f) => f.type.startsWith('image/'))
   152→   152→    if (images.length < picked.length) toast.error('Only image files can be added')
   153→   153→    if (images.length === 0) {
   154→   154→      resetFileInput()
   155→   155→      return
   156→   156→    }
   157→   157→    const batch = images.slice(0, remaining)
   158→   158→    if (images.length > remaining) {
   159→   159→      toast.warning(`Only ${remaining} more image${remaining === 1 ? '' : 's'} allowed`)
   160→   160→    }
   161→   161→    setProcessing(true)

   163→   163→      const added: string[] = []

   166→   166→          added.push(await fileToCompressedDataUrl(file))
   167→   167→        } catch {
   168→   168→          toast.error(`Could not process "${file.name}"`)
   169→   169→        }
   170→   170→      }
   171→   171→      if (added.length > 0) onChange([...value, ...added])
   172→   172→    } finally {
   173→   173→      setProcessing(false)
   174→   174→      resetFileInput()
   175→   175→    }

   282→   282→function validate(f: FormState): string | null {
   283→   283→  if (!f.name.trim()) return 'Medicine name is required'
   284→   284→  const price = Number(f.price)
   285→   285→  if (!f.price || Number.isNaN(price) || price <= 0) return 'Price must be greater than 0'
   286→   286→  const stock = Number(f.stock)
   287→   287→  if (f.stock === '' || Number.isNaN(stock) || stock < 0) return 'Stock must be 0 or more'
   288→   288→  if (f.discountPrice.trim() !== '') {
   289→   289→    const d = Number(f.discountPrice)
   290→   290→    if (Number.isNaN(d) || d < 0) return 'Discount price must be 0 or more'
   291→   291→    if (d >= price) return 'Discount price must be lower than the regular price'
   292→   292→  }
   293→   293→  return null
   294→   294→}
