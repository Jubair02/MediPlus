    49→/** 1dp when fractional, plain integer otherwise (100 → "100", 2.1 → "2.1"). */

    51→  return Number.isInteger(n) ? String(n) : n.toFixed(1)
    52→}

    54→function daysLeftLabel(daysLeft: number | null): string {

    59→/** Today as a local YYYY-MM-DD string (used as the date input min). */
    60→function todayLocal(): string {
    61→  const d = new Date()
    62→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`

   128→/** Plain-text PO draft for suppliers — mirrors the table rows. */
   129→export function buildPoDraft(list: RestockSuggestion[]): string {
   130→  const date = new Date().toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' })
   131→  const lines = list.map((s, i) => {
   132→    const days = s.daysLeft != null ? `~${fmtNum(s.daysLeft)}d left` : 'no recent sales'
   133→    const brand = s.brand ? ` (${s.brand})` : ''
   134→    return `${i + 1}. ${s.name}${brand} (${s.unit}) — suggested ${s.suggestedQty} (stock ${s.stock}, ${days}) — est ${fmtBDT(s.estValue)}`
   135→  })
   136→  return ['MediPlus restock draft — ' + date, ...lines].join('\n')
   137→}

   163→  const [poSubmitting, setPoSubmitting] = useState(false)
   164→

   166→    if (silent) setRefreshing(true)
   167→    else {
   168→      setLoading(true)

   178→      if (silent) toast.error(message)
   179→      else setError(message)
   180→    } finally {
   181→      setLoading(false)
   182→      setRefreshing(false)
   183→    }
   184→  }, [])
   185→
   186→  useEffect(() => {
   187→    void load()
   188→  }, [load])

   194→      spend: list.reduce((sum, s) => sum + s.estValue, 0),

   218→  function closePoDialog() {

   227→  async function submitPo(): Promise<void> {
   228→    const s = poTarget
   229→    if (!s || poSubmitting) return
   230→    const qty = Number(poQty)
   231→    if (poQty.trim() === '' || !Number.isInteger(qty) || qty < 1 || qty > 10000) {
   232→      setPoError('Quantity must be between 1 and 10000')
   233→      return
   234→    }
   235→    setPoError(null)
   236→    setPoSubmitting(true)
   237→    setOrderingIds((prev) => {
   238→      const next = new Set(prev)
   239→      next.add(s.id)
   240→      return next
   241→    })
   242→    try {
   243→      const supplier = poSupplier.trim()
   244→      const note = poNote.trim()
   245→      await api<{ order: unknown }>('/api/pharmacist', {
   246→        method: 'PUT',
   247→        body: {
   248→          action: 'create-po',
   249→          medicineId: s.id,
   250→          qty,
   251→          ...(supplier !== '' ? { supplier } : {}),
   252→          ...(poExpectedAt !== '' ? { expectedAt: poExpectedAt } : {}),
   253→          ...(note !== '' ? { note } : {}),
   254→        },
   255→      })
   256→      toast.success(`PO created — ${qty} × ${s.name}`)
   257→      closePoDialog()
   258→      onStatsChanged?.()
   259→      await load(true) // silent refresh so openPoQty / coverage chips update
   260→    } catch (e) {
   261→      // Server 400 (e.g. supplier/date validation) — keep the dialog open
   262→      toast.error(e instanceof Error ? e.message : 'Failed to create purchase order')
   263→    } finally {
   264→      setPoSubmitting(false)
   265→      setOrderingIds((prev) => {
   266→        const next = new Set(prev)
   267→        next.delete(s.id)
   268→        return next
   269→      })
   270→    }
   271→  }

   273→  async function orderAll() {
   274→    if (needOrdering.length === 0) return
   275→    setOrderAllPending(true)
   276→    let ok = 0
   277→    let failed = 0
   278→    let firstError: string | null = null
   279→    try {
   280→      for (const s of needOrdering) {
   281→        try {
   282→          await api<{ order: unknown }>('/api/pharmacist', {
   283→            method: 'PUT',
   284→            body: { action: 'create-po', medicineId: s.id, qty: s.suggestedQty },
   285→          })
   286→          ok++
   287→        } catch (e) {
   288→          failed++
   289→          if (!firstError) firstError = e instanceof Error ? e.message : 'Unknown error'
   290→        }
   291→      }
   292→      if (ok > 0) {
   293→        toast.success(`Created ${ok} purchase order${ok === 1 ? '' : 's'}${failed > 0 ? ` — ${failed} failed` : ''}`)
   294→        onStatsChanged?.()
   295→        await load(true)
   296→      }
   297→      if (failed > 0) {
   298→        toast.error(`Failed ${failed} order${failed === 1 ? '' : 's'} — ${firstError ?? 'unknown error'}`)
   299→        await load(true)
   300→      }
   301→    } finally {
   302→      setOrderAllPending(false)
   303→      setOrderAllConfirm(false)
   304→    }
   305→  }
