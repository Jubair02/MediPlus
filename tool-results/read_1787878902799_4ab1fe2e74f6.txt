   102→   102→async function restockSuggestions() {
   103→   103→  const since = new Date(Date.now() - RESTOCK_WINDOW_DAYS * 24 * 60 * 60 * 1000)
   104→   104→  const medicines = await db.medicine.findMany({
   105→   105→    where: { status: 'ACTIVE', stock: { lte: LOW_STOCK_THRESHOLD } },
   106→   106→    orderBy: { stock: 'asc' },
   107→   107→  })
   108→   108→  const ids = medicines.map((m) => m.id)
   109→   109→  const soldRows = ids.length
   110→   110→    ? await db.orderItem.findMany({
   111→   111→        where: {
   112→   112→          medicineId: { in: ids },
   113→   113→          order: { status: { notIn: ['CANCELLED', 'FAILED'] }, createdAt: { gte: since } },
   114→   114→        },
   115→   115→        select: { medicineId: true, quantity: true },
   116→   116→      })
   117→   117→    : []
   118→   118→  const soldByMed = new Map<string, number>()
   119→   119→  for (const row of soldRows) {
   120→   120→    if (row.medicineId) soldByMed.set(row.medicineId, (soldByMed.get(row.medicineId) ?? 0) + row.quantity)
   121→   121→  }
   122→   122→  // Round 10 — total qty across each medicine's OPEN (ORDERED) purchase orders
   123→   123→  const openPoGroups = ids.length
   124→   124→    ? await db.purchaseOrder.groupBy({ by: ['medicineId'], where: { medicineId: { in: ids }, status: 'ORDERED' }, _sum: { qty: true } })
   125→   125→    : []
   126→   126→  const openPoByMed = new Map(openPoGroups.map((g) => [g.medicineId, g._sum.qty ?? 0]))
   127→   127→
   128→   128→  const suggestions = medicines.map((m) => {
   129→   129→    const soldLast30 = soldByMed.get(m.id) ?? 0
   130→   130→    const avgDaily = round1(soldLast30 / RESTOCK_WINDOW_DAYS)
   131→   131→    let daysLeft: number | null
   132→   132→    if (m.stock <= 0) daysLeft = 0
   133→   133→    else if (avgDaily <= 0) daysLeft = null
   134→   134→    else daysLeft = round1(m.stock / avgDaily)
   135→   135→    const base = Math.max(30, Math.ceil(soldLast30 * 1.5)) - m.stock
   136→   136→    let suggestedQty = Math.ceil(Math.max(base, 10) / 10) * 10
   137→   137→    if (m.stock <= 0) suggestedQty = Math.max(suggestedQty, 30)
   138→   138→    return {
   139→   139→      id: m.id,
   140→   140→      name: m.name,
   141→   141→      brand: m.brand,
   142→   142→      genericName: m.genericName,
   143→   143→      unit: m.unit,
   144→   144→      image: m.image,
   145→   145→      stock: m.stock,
   146→   146→      lowStockAt: LOW_STOCK_THRESHOLD,
   147→   147→      soldLast30,
   148→   148→      avgDaily,
   149→   149→      daysLeft,
   150→   150→      suggestedQty,
   151→   151→      estValue: round2(suggestedQty * (m.discountPrice ?? m.price)),
   152→   152→      openPoQty: openPoByMed.get(m.id) ?? 0,
   153→   153→    }
   154→   154→  })
   155→   155→  // daysLeft asc (nulls last), then stock asc
   156→   156→  suggestions.sort((a, b) => {
   157→   157→    const an = a.daysLeft === null ? 1 : 0
   158→   158→    const bn = b.daysLeft === null ? 1 : 0
   159→   159→    if (an !== bn) return an - bn
   160→   160→    if (a.daysLeft !== null && b.daysLeft !== null && a.daysLeft !== b.daysLeft) return a.daysLeft - b.daysLeft
   161→   161→    return a.stock - b.stock
   162→   162→  })
   163→   163→  return { suggestions }
   164→   164→}

   364→   364→/**
   365→   365→ * PUT /api/pharmacist
   366→   366→ *  {action:'review', prescriptionId, decision:'APPROVED'|'REJECTED', reviewNote?}  — sets reviewedAt + RX_REVIEW audit entry
   367→   367→ *  {action:'create-po', medicineId, qty, note?, supplier?, expectedAt?} — create purchase order (PO_CREATE audit entry);
   368→   368→ *      supplier trimmed, empty → null, ≤120 chars else 400 'Supplier must be 120 characters or less'; expectedAt
   369→   369→ *      'YYYY-MM-DD' (local midnight) or ISO string, NaN → 400 'Invalid expected date', before today → 400
   370→   370→ *      'Expected date cannot be in the past'
   371→   371→ *  {action:'receive-po', id, note?}                  — receive PO: stock +qty, StockMovement PO_RECEIVE, audit entry
   372→   372→ *  {action:'cancel-po', id}                          — cancel an ORDERED purchase order (PO_CANCEL audit entry)
   373→   373→ *  {action:'create-medicine', data:{...}}            — data.extraImages (optional, ≤5 data:/http(s): urls) persisted with sort = index
   374→   374→ *  {action:'update-medicine', id, data:{...}}        — data.extraImages present = REPLACE-ALL (empty array clears); absent = untouched
   375→   375→ *  {action:'answer', id, answer}   — answer a Q&A question (re-answer allowed, overwrites)
   376→   376→ *  {action:'reject', id}           — reject a Q&A question (answer/answeredBy/answeredAt cleared)
   377→   377→ */
   378→   378→export async function PUT(request: Request) {
   379→   379→  try {
   380→   380→    const user = await requireRole(request, ['PHARMACIST'])
   381→   381→    if (user instanceof Response) return user
   382→   382→    const body = await readJson(request)
   383→   383→    if (!body) return badRequest('Invalid request body')
   384→   384→    const action = optStr(body.action)
   385→   385→
   386→   386→    if (action === 'review') {
   387→   387→      const prescriptionId = optStr(body.prescriptionId)
   388→   388→      if (!prescriptionId) return badRequest('Prescription id is required')
   389→   389→      const decision = optStr(body.decision)
   390→   390→      if (decision !== 'APPROVED' && decision !== 'REJECTED') return badRequest('Decision must be APPROVED or REJECTED')
   391→   391→
   392→   392→      const prescription = await db.prescription.findUnique({
   393→   393→        where: { id: prescriptionId },
   394→   394→        include: { order: { include: { ...orderInclude, items: { include: { medicine: true } } } } },
   395→   395→      })
   396→   396→      if (!prescription) return notFound('Prescription not found')
   397→   397→      if (prescription.status !== 'PENDING') return badRequest('Already reviewed')
   398→   398→
   399→   399→      const reviewNote = optStr(body.reviewNote)?.trim() || null
   400→   400→      const order = prescription.order
   401→   401→
   402→   402→      if (decision === 'APPROVED' && order) {
   403→   403→        // Check stock for all Rx order items BEFORE approving
   404→   404→        for (const item of order.items) {
   405→   405→          if (!item.medicine) return badRequest(`Insufficient stock for ${item.name}`)
   406→   406→          if (item.medicine.status !== 'ACTIVE' || item.medicine.stock < item.quantity) {
   407→   407→            return badRequest(`Insufficient stock for ${item.name}`)
   408→   408→          }
   409→   409→        }
   410→   410→        await db.$transaction(async (tx) => {
   411→   411→          await tx.prescription.update({
   412→   412→            where: { id: prescription.id },
   413→   413→            data: { status: 'APPROVED', reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   414→   414→          })
   415→   415→          const movements: StockMovementEntry[] = []
   416→   416→          for (const item of order.items) {
   417→   417→            if (item.medicineId) {
   418→   418→              await tx.medicine.update({ where: { id: item.medicineId }, data: { stock: { decrement: item.quantity } } })
   419→   419→              movements.push({
   420→   420→                medicineId: item.medicineId,
   421→   421→                delta: -item.quantity,
   422→   422→                reason: 'RX_APPROVE',
   423→   423→                note: `Order ${order.orderNo} confirmed after prescription approval`,
   424→   424→                userId: user.id,
   425→   425→              })
   426→   426→            }
   427→   427→          }
   428→   428→          await recordStockMovements(tx, movements)
   429→   429→          await tx.order.update({ where: { id: order.id }, data: { status: 'CONFIRMED' } })
   430→   430→          const existingPayment = await tx.payment.findUnique({ where: { orderId: order.id } })
   431→   431→          if (!existingPayment) {
   432→   432→            await tx.payment.create({
   433→   433→              data: {
   434→   434→                orderId: order.id,
   435→   435→                method: order.paymentMethod,
   436→   436→                status: order.paymentStatus === 'PAID' ? 'PAID' : 'PENDING',
   437→   437→                amount: order.total,
   438→   438→              },
   439→   439→            })
   440→   440→          }
   441→   441→          await tx.notification.create({
   442→   442→            data: { userId: order.userId, title: 'Prescription approved', message: 'Your prescription has been approved.' },
   443→   443→          })
   444→   444→          await tx.notification.create({
   445→   445→            data: { userId: order.userId, title: 'Order confirmed', message: `Order ${order.orderNo} has been confirmed and is being processed.` },
   446→   446→          })
   447→   447→          await logAudit(tx, user, {
   448→   448→            action: 'RX_REVIEW',
   449→   449→            entityType: 'PRESCRIPTION',
   450→   450→            entityRef: prescription.id,
   451→   451→            detail: `Prescription ${decision === 'APPROVED' ? 'approved' : 'rejected'}`,
   452→   452→          })
   453→   453→        })
   454→   454→      } else if (decision === 'REJECTED' && order) {
   455→   455→        await db.$transaction(async (tx) => {
   456→   456→          await tx.prescription.update({
   457→   457→            where: { id: prescription.id },
   458→   458→            data: { status: 'REJECTED', reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   459→   459→          })
   460→   460→          await tx.order.update({ where: { id: order.id }, data: { status: 'CANCELLED', statusNote: 'Prescription rejected' } })
   461→   461→          await tx.notification.create({
   462→   462→            data: {
   463→   463→              userId: order.userId,
   464→   464→              title: 'Prescription rejected',
   465→   465→              message: `Your prescription was rejected.${reviewNote ? ` Reason: ${reviewNote}` : ''}`,
   466→   466→            },
   467→   467→          })
   468→   468→          await logAudit(tx, user, {
   469→   469→            action: 'RX_REVIEW',
   470→   470→            entityType: 'PRESCRIPTION',
   471→   471→            entityRef: prescription.id,
   472→   472→            detail: 'Prescription rejected',
   473→   473→          })
   474→   474→        })
   475→   475→      } else {
   476→   476→        await db.prescription.update({
   477→   477→          where: { id: prescription.id },
   478→   478→          data: { status: decision, reviewNote, reviewedById: user.id, reviewedAt: new Date() },
   479→   479→        })
   480→   480→        await logAudit(db, user, {
   481→   481→          action: 'RX_REVIEW',
   482→   482→          entityType: 'PRESCRIPTION',
   483→   483→          entityRef: prescription.id,
   484→   484→          detail: `Prescription ${decision === 'APPROVED' ? 'approved' : 'rejected'}`,
   485→   485→        })
   486→   486→        await notify(
   487→   487→          prescription.userId,
   488→   488→          decision === 'APPROVED' ? 'Prescription approved' : 'Prescription rejected',
   489→   489→          decision === 'APPROVED'
   490→   490→            ? 'Your prescription has been approved.'
   491→   491→            : `Your prescription was rejected.${reviewNote ? ` Reason: ${reviewNote}` : ''}`
   492→   492→        )
   493→   493→      }
   494→   494→
   495→   495→      const updated = await db.prescription.findUnique({
   496→   496→        where: { id: prescription.id },
   497→   497→        include: { order: { select: { orderNo: true } } },
   498→   498→      })
   499→   499→      if (!updated) return notFound('Prescription not found')
   500→   500→      return Response.json({
   501→   501→        prescription: {
   502→   502→          id: updated.id,
   503→   503→          image: updated.image,
   504→   504→          note: updated.note,
   505→   505→          status: updated.status,
   506→   506→          reviewNote: updated.reviewNote,
   507→   507→          createdAt: updated.createdAt,
   508→   508→          orderNo: updated.order?.orderNo ?? null,
   509→   509→        },
   510→   510→      })
   511→   511→    }
   512→   512→
   513→   513→    if (action === 'create-po') {
   514→   514→      const medicineId = optStr(body.medicineId)?.trim()
   515→   515→      if (!medicineId) return badRequest('Medicine id is required')
   516→   516→      const medicine = await db.medicine.findUnique({ where: { id: medicineId } })
   517→   517→      if (!medicine) return notFound('Medicine not found')
   518→   518→      const qty = numOr(body.qty, NaN)
   519→   519→      if (!Number.isInteger(qty) || qty < 1 || qty > PO_MAX_QTY) return badRequest('Quantity must be between 1 and 10000')
   520→   520→      let note: string | null = null
   521→   521→      if (typeof body.note === 'string' && body.note.trim() !== '') {
   522→   522→        note = body.note.trim()
   523→   523→        if (note.length > PO_MAX_NOTE_LENGTH) return badRequest('Note must be 300 characters or less')
   524→   524→      }
   525→   525→      // Round 11 — optional supplier (trimmed, empty → null) + expected delivery date
   526→   526→      let supplier: string | null = null
   527→   527→      if (typeof body.supplier === 'string' && body.supplier.trim() !== '') {
   528→   528→        supplier = body.supplier.trim()
   529→   529→        if (supplier.length > PO_MAX_SUPPLIER_LENGTH) return badRequest('Supplier must be 120 characters or less')
   530→   530→      }
   531→   531→      let expectedAt: Date | null = null
   532→   532→      if (typeof body.expectedAt === 'string' && body.expectedAt.trim() !== '') {
   533→   533→        const raw = body.expectedAt.trim()
   534→   534→        // 'YYYY-MM-DD' is treated as local midnight; anything else parses as an ISO/date string
   535→   535→        const dt = /^\d{4}-\d{2}-\d{2}$/.test(raw)
   536→   536→          ? (() => {
   537→   537→              const [y, m, d] = raw.split('-').map(Number)
   538→   538→              return new Date(y, m - 1, d)
   539→   539→            })()
   540→   540→          : new Date(raw)
   541→   541→        if (Number.isNaN(dt.getTime())) return badRequest('Invalid expected date')
   542→   542→        const todayStart = new Date()
   543→   543→        todayStart.setHours(0, 0, 0, 0)
   544→   544→        if (dt.getTime() < todayStart.getTime()) return badRequest('Expected date cannot be in the past')
   545→   545→        expectedAt = dt
   546→   546→      }
   547→   547→      const created = await db.purchaseOrder.create({
   548→   548→        data: { medicineId: medicine.id, qty, note, supplier, expectedAt, orderedById: user.id },
   549→   549→      })
   550→   550→      await logAudit(db, user, {
   551→   551→        action: 'PO_CREATE',
   552→   552→        entityType: 'PURCHASE_ORDER',
   553→   553→        entityRef: created.id,
   554→   554→        detail: `Ordered ${qty} × ${medicine.name}`,
   555→   555→      })
   556→   556→      const fresh = await db.purchaseOrder.findUnique({ where: { id: created.id }, include: poInclude })
   557→   557→      if (!fresh) return notFound('Purchase order not found')
   558→   558→      return Response.json({ order: poToRow(fresh) })
   559→   559→    }
   560→   560→
   561→   561→    if (action === 'receive-po') {
   562→   562→      const id = optStr(body.id)
   563→   563→      if (!id) return badRequest('Purchase order id is required')
   564→   564→      const po = await db.purchaseOrder.findUnique({
   565→   565→        where: { id },
   566→   566→        include: { medicine: { select: { id: true, name: true } } },
   567→   567→      })
   568→   568→      if (!po) return notFound('Purchase order not found')
   569→   569→      if (po.status !== 'ORDERED') return badRequest('Only ordered purchase orders can be received')
   570→   570→      // body.note replaces the note only when provided as a non-empty string; otherwise the note is kept
   571→   571→      const replacementNote = typeof body.note === 'string' && body.note.trim() !== '' ? body.note.trim() : null
   572→   572→      await db.$transaction(async (tx) => {
   573→   573→        await tx.purchaseOrder.update({
   574→   574→          where: { id: po.id },
   575→   575→          data: {
   576→   576→            status: 'RECEIVED',
   577→   577→            receivedAt: new Date(),
   578→   578→            receivedById: user.id,
   579→   579→            ...(replacementNote !== null ? { note: replacementNote } : {}),
   580→   580→          },
   581→   581→        })
   582→   582→        await tx.medicine.update({ where: { id: po.medicineId }, data: { stock: { increment: po.qty } } })
   583→   583→        await recordStockMovements(tx, [{
   584→   584→          medicineId: po.medicineId,
   585→   585→          delta: po.qty,
   586→   586→          reason: 'PO_RECEIVE',
   587→   587→          note: `PO received — ${po.qty} × ${po.medicine.name}`,
   588→   588→          userId: user.id,
   589→   589→        }])
   590→   590→        await logAudit(tx, user, {
   591→   591→          action: 'PO_RECEIVE',
   592→   592→          entityType: 'PURCHASE_ORDER',
   593→   593→          entityRef: po.id,
   594→   594→          detail: `Received ${po.qty} × ${po.medicine.name}`,
   595→   595→        })
   596→   596→      })
   597→   597→      const fresh = await db.purchaseOrder.findUnique({ where: { id: po.id }, include: poInclude })
   598→   598→      const medicine = await db.medicine.findUnique({ where: { id: po.medicineId }, select: { id: true, stock: true } })
   599→   599→      if (!fresh || !medicine) return notFound('Purchase order not found')
   600→   600→      return Response.json({ order: poToRow(fresh), medicine: { id: medicine.id, stock: medicine.stock } })
   601→   601→    }
   602→   602→
   603→   603→    if (action === 'cancel-po') {
   604→   604→      const id = optStr(body.id)
   605→   605→      if (!id) return badRequest('Purchase order id is required')
   606→   606→      const po = await db.purchaseOrder.findUnique({
   607→   607→        where: { id },
   608→   608→        include: { medicine: { select: { name: true } } },
   609→   609→      })
   610→   610→      if (!po) return notFound('Purchase order not found')
   611→   611→      if (po.status !== 'ORDERED') return badRequest('Only ordered purchase orders can be cancelled')
   612→   612→      await db.purchaseOrder.update({ where: { id: po.id }, data: { status: 'CANCELLED' } })
   613→   613→      await logAudit(db, user, {
   614→   614→        action: 'PO_CANCEL',
   615→   615→        entityType: 'PURCHASE_ORDER',
   616→   616→        entityRef: po.id,
   617→   617→        detail: `Cancelled ${po.qty} × ${po.medicine.name}`,
   618→   618→      })
   619→   619→      const fresh = await db.purchaseOrder.findUnique({ where: { id: po.id }, include: poInclude })
   620→   620→      if (!fresh) return notFound('Purchase order not found')
   621→   621→      return Response.json({ order: poToRow(fresh) })
   622→   622→    }
   623→   623→
   624→   624→    if (action === 'create-medicine') {
   625→   625→      const parsed = await buildMedicineData(body.data, 'create')
   626→   626→      if ('error' in parsed) return badRequest(parsed.error)
   627→   627→      const medicine = await db.medicine.create({
   628→   628→        data: parsed.data,
   629→   629→        include: { category: true, extraImages: { orderBy: { sort: 'asc' } } },
   630→   630→      })
   631→   631→      if (medicine.stock > 0) {
   632→   632→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: medicine.stock, reason: 'MANUAL_EDIT', note: 'Initial stock', userId: user.id }])
   633→   633→      }
   634→   634→      return Response.json({ medicine: medicineStaffJson(medicine) }, { status: 201 })
   635→   635→    }
   636→   636→
   637→   637→    if (action === 'update-medicine') {
   638→   638→      const id = optStr(body.id)
   639→   639→      if (!id) return badRequest('Medicine id is required')
   640→   640→      const existing = await db.medicine.findUnique({ where: { id } })
   641→   641→      if (!existing) return notFound('Medicine not found')
   642→   642→      const parsed = await buildMedicineData(body.data, 'update')
   643→   643→      if ('error' in parsed) return badRequest(parsed.error)
   644→   644→      const newStock = typeof parsed.data.stock === 'number' ? parsed.data.stock : undefined
   645→   645→      const medicine = await db.medicine.update({
   646→   646→        where: { id },
   647→   647→        data: parsed.data,
   648→   648→        include: { category: true, extraImages: { orderBy: { sort: 'asc' } } },
   649→   649→      })
   650→   650→      if (newStock !== undefined && newStock !== existing.stock) {
   651→   651→        await recordStockMovements(db, [{ medicineId: medicine.id, delta: newStock - existing.stock, reason: 'MANUAL_EDIT', note: 'Manual stock update', userId: user.id }])
   652→   652→      }
   653→   653→      return Response.json({ medicine: medicineStaffJson(medicine) })
   654→   654→    }
   655→   655→
   656→   656→    if (action === 'answer' || action === 'reject') {
   657→   657→      const id = optStr(body.id)
   658→   658→      if (!id) return badRequest('Question id is required')
   659→   659→      const existing = await db.question.findUnique({ where: { id } })
   660→   660→      if (!existing) return notFound('Question not found')
   661→   661→
   662→   662→      if (action === 'answer') {
   663→   663→        const answer = typeof body.answer === 'string' ? body.answer.trim() : ''
   664→   664→        if (answer.length < 2) return badRequest('Answer must be at least 2 characters')
   665→   665→        if (answer.length > MAX_ANSWER_LENGTH) return badRequest(`Answer must be ${MAX_ANSWER_LENGTH} characters or less`)
   666→   666→        await db.question.update({
   667→   667→          where: { id },
   668→   668→          data: { answer, status: 'ANSWERED', answeredById: user.id, answeredAt: new Date() },
   669→   669→        })
   670→   670→        // Notify the asker so they come back to read the answer (skip on re-answer of same content)
   671→   671→        const medicineName = existing.medicineId
   672→   672→          ? (await db.medicine.findUnique({ where: { id: existing.medicineId }, select: { name: true } }))?.name ?? 'a medicine'
   673→   673→          : 'a medicine'
   674→   674→        await notify(
   675→   675→          existing.userId,
   676→   676→          'Your question was answered',
   677→   677→          `A pharmacist answered your question about ${medicineName}: "${answer.slice(0, 120)}${answer.length > 120 ? '…' : ''}"`
   678→   678→        )
   679→   679→      } else {
   680→   680→        await db.question.update({
   681→   681→          where: { id },
   682→   682→          data: { status: 'REJECTED', answer: null, answeredById: null, answeredAt: null },
   683→   683→        })
   684→   684→      }
   685→   685→
   686→   686→      const updated = await db.question.findUnique({ where: { id }, include: questionInclude })
   687→   687→      if (!updated) return notFound('Question not found')
   688→   688→      return Response.json({ question: questionToJson(updated) })
   689→   689→    }
   690→   690→
   691→   691→    return badRequest('Invalid action')
   692→   692→  } catch (e) {
   693→   693→    return serverError(e)
   694→   694→  }
   695→   695→}
