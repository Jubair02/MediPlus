    91→/**
    92→ * The API contract requires every extra image to start with `data:image` or http(s):// —
    93→ * origin-relative paths (e.g. seeded '/images/med-napa.png') are normalized to absolute
    94→ * URLs here so seeded rows round-trip through update-medicine unchanged in appearance.

    96→function normalizeImageSrc(src: string): string {
    97→  const s = src.trim()
    98→  if (s.startsWith('/') && typeof window !== 'undefined') return `${window.location.origin}${s}`
    99→  return s
   100→}
   101→
   102→function isValidImageSrc(src: string): boolean {
   103→  const s = normalizeImageSrc(src)
   104→  return s.startsWith('data:image') || /^https?:\/\//i.test(s)
   105→}

   107→/**
   108→ * Round 11 — extra-image manager: thumbnail grid with remove buttons, file upload
   109→ * (compressed via fileToCompressedDataUrl) and URL add. Hard cap of MAX_EXTRA_IMAGES.
   110→ */
   111→function ExtraImagesField({
   112→  value,
   113→  onChange,
   114→}: {
   115→  value: string[]
   116→  onChange: (next: string[]) => void
   117→}) {
   118→  const [urlInput, setUrlInput] = useState('')
   119→  const [processing, setProcessing] = useState(false)
   120→  const fileRef = useRef<HTMLInputElement>(null)
   121→
   122→  const remaining = MAX_EXTRA_IMAGES - value.length
   123→  const atCap = remaining <= 0
   124→
   125→  function resetFileInput() {
   126→    if (fileRef.current) fileRef.current.value = ''
   127→  }
   128→
   129→  function addUrl() {
   130→    const src = urlInput.trim()
   131→    if (!src || processing) return
   132→    if (!isValidImageSrc(src)) {
   133→      toast.error('Image must be an http(s) URL or an image data URL')
   134→      return
   135→    }
   136→    onChange([...value, normalizeImageSrc(src)])
   137→    setUrlInput('')
   138→  }
   139→
   140→  async function onFilesPicked(files: FileList | null) {
   141→    if (!files || files.length === 0 || processing) {
   142→      resetFileInput()
   143→      return
   144→    }
   145→    if (atCap) {
   146→      toast.warning(`Only ${remaining} more image${remaining === 1 ? '' : 's'} allowed`)
   147→      resetFileInput()
   148→      return
   149→    }
   150→    const picked = Array.from(files)
   151→    const images = picked.filter((f) => f.type.startsWith('image/'))
   152→    if (images.length < picked.length) toast.error('Only image files can be added')
   153→    if (images.length === 0) {
   154→      resetFileInput()
   155→      return
   156→    }
   157→    const batch = images.slice(0, remaining)
   158→    if (images.length > remaining) {
   159→      toast.warning(`Only ${remaining} more image${remaining === 1 ? '' : 's'} allowed`)
   160→    }
   161→    setProcessing(true)
   162→    try {
   163→      const added: string[] = []
   164→      for (const file of batch) {
   165→        try {
   166→          added.push(await fileToCompressedDataUrl(file))
   167→        } catch {
   168→          toast.error(`Could not process "${file.name}"`)
   169→        }
   170→      }
   171→      if (added.length > 0) onChange([...value, ...added])
   172→    } finally {
   173→      setProcessing(false)
   174→      resetFileInput()
   175→    }
   176→  }
   177→
   178→  function removeAt(index: number) {
   179→    if (processing) return
   180→    onChange(value.filter((_, i) => i !== index))
   181→  }
   182→
   183→  return (
   184→    <div className="space-y-2.5 rounded-lg border p-3 sm:col-span-2">
   185→      <div className="flex flex-wrap items-start justify-between gap-2">
   186→        <div className="space-y-0.5">
   187→          <Label>Extra images</Label>
   188→          <p className="text-[11px] text-muted-foreground">Shown as a gallery on the product page</p>
   189→        </div>
   190→        <span className="text-[11px] text-muted-foreground tabular-nums">
   191→          {value.length} / {MAX_EXTRA_IMAGES} images · up to {MAX_EXTRA_IMAGES} extra images
   192→        </span>
   193→      </div>
   194→
   195→      {value.length > 0 && (
   196→        <div className="flex flex-wrap gap-2">
   197→          {value.map((src, i) => (
   198→            <div key={`extra-${i}`} className="relative">
   199→              <MedImage
   200→                src={src}
   201→                alt={`Extra image ${i + 1}`}
   202→                className="h-20 w-20 rounded-lg border border-border bg-muted object-cover"
   203→              />
   204→              <button
   205→                type="button"
   206→                onClick={() => removeAt(i)}
   207→                disabled={processing}
   208→                className="absolute top-1 right-1 flex size-5 items-center justify-center rounded-full border bg-background/90 text-foreground transition-colors hover:bg-destructive hover:text-destructive-foreground focus-visible:ring-primary/30 focus-visible:outline-none focus-visible:ring-2 disabled:pointer-events-none disabled:opacity-50"
   209→                aria-label={`Remove image ${i + 1}`}
   210→                title={`Remove image ${i + 1}`}
   211→              >
   212→                <X className="size-3" />
   213→              </button>
   214→            </div>
   215→          ))}
   216→        </div>
   217→      )}
   218→
   219→      {/* Add paths — always rendered so the cap state reads as disabled (not missing) */}
   220→      <div className="flex flex-col gap-2 sm:flex-row">
   221→        <input
   222→          ref={fileRef}
   223→          type="file"
   224→          accept="image/*"
   225→          multiple
   226→          className="sr-only"
   227→          tabIndex={-1}
   228→          aria-hidden="true"
   229→          onChange={(e) => void onFilesPicked(e.target.files)}
   230→        />
   231→        <Button
   232→          type="button"
   233→          variant="outline"
   234→          size="sm"
   235→          className="h-9 shrink-0 focus-visible:ring-primary/30"
   236→          disabled={processing || atCap}
   237→          onClick={() => fileRef.current?.click()}
   238→          title="Upload image files (compressed automatically)"
   239→          aria-label="Upload extra images"
   240→        >
   241→          {processing ? <Loader2 className="h-4 w-4 animate-spin" /> : <ImagePlus className="h-4 w-4" />}
   242→          {processing ? 'Processing…' : 'Upload'}
   243→        </Button>
   244→        <div className="flex flex-1 items-center gap-2">
   245→          <Input
   246→            value={urlInput}
   247→            onChange={(e) => setUrlInput(e.target.value)}
   248→            onKeyDown={(e) => {
   249→              if (e.key === 'Enter') {
   250→                e.preventDefault()
   251→                addUrl()
   252→              }
   253→            }}
   254→            placeholder="https://… or paste image URL"
   255→            aria-label="Extra image URL"
   256→            disabled={processing || atCap}
   257→            className="h-9 flex-1"
   258→          />
   259→          <Button
   260→            type="button"
   261→            variant="outline"
   262→            size="sm"
   263→            className="h-9 shrink-0 focus-visible:ring-primary/30"
   264→            disabled={processing || atCap || urlInput.trim() === ''}
   265→            onClick={addUrl}
   266→            title="Add image URL"
   267→          >
   268→            <Plus className="h-4 w-4" />
   269→            Add
   270→          </Button>
   271→        </div>
   272→      </div>
   273→      {atCap && (
   274→        <p className="text-[11px] text-muted-foreground">
   275→          Maximum of {MAX_EXTRA_IMAGES} extra images — remove one to add another.
   276→        </p>
   277→      )}
   278→    </div>
   279→  )

   282→function validate(f: FormState): string | null {
   283→  if (!f.name.trim()) return 'Medicine name is required'
   284→  const price = Number(f.price)
   285→  if (!f.price || Number.isNaN(price) || price <= 0) return 'Price must be greater than 0'
   286→  const stock = Number(f.stock)
   287→  if (f.stock === '' || Number.isNaN(stock) || stock < 0) return 'Stock must be 0 or more'
   288→  if (f.discountPrice.trim() !== '') {
   289→    const d = Number(f.discountPrice)
   290→    if (Number.isNaN(d) || d < 0) return 'Discount price must be 0 or more'
   291→    if (d >= price) return 'Discount price must be lower than the regular price'
   292→  }
   293→  return null
   294→}

   374→    } catch (e) {
   375→      toast.error(e instanceof Error ? e.message : 'Failed to save medicine')

   378→    }

   579→      setMeds(m.medicines)
   580→      setCategories(c.categories)
   581→    } catch (e) {
   582→      toast.error(e instanceof Error ? e.message : 'Failed to load medicines')
   583→    } finally {
   584→      setLoading(false)
   585→    }
   586→  }, [])
   587→
   588→  useEffect(() => {
   589→    void load()
   590→  }, [load])
   591→
   592→  const filtered = useMemo(() => {
   593→    const q = search.trim().toLowerCase()
   594→    if (!q) return meds
   595→    return meds.filter((m) =>
   596→      `${m.name} ${m.genericName ?? ''} ${m.brand ?? ''}`.toLowerCase().includes(q)
   597→    )
   598→  }, [meds, search])
   599→
   600→  async function adjustStock(m: Medicine, delta: number) {
   601→    const next = Math.max(0, m.stock + delta)
   602→    if (next === m.stock) return
   603→    setRowPending(m.id)
   604→    try {
   605→      const d = await api<{ medicine: Medicine }>('/api/pharmacist', {
   606→        method: 'PUT',
   607→        body: { action: 'update-medicine', id: m.id, data: { stock: next } },
   608→      })
   609→      setMeds((prev) => prev.map((x) => (x.id === m.id ? d.medicine : x)))
   610→      toast.success(`${m.name}: stock updated to ${d.medicine.stock}`)
   611→    } catch (e) {
   612→      toast.error(e instanceof Error ? e.message : 'Failed to update stock')
   613→    } finally {
   614→      setRowPending(null)
   615→    }
   616→  }
   617→
   618→  async function toggleStatus(m: Medicine, next: boolean) {
   619→    setRowPending(m.id)
   620→    try {
   621→      await api('/api/pharmacist', {
   622→        method: 'PUT',
   623→        body: { action: 'update-medicine', id: m.id, data: { status: next ? 'ACTIVE' : 'INACTIVE' } },
   624→      })
   625→      setMeds((prev) =>
   626→        prev.map((x) => (x.id === m.id ? { ...x, status: next ? 'ACTIVE' : 'INACTIVE' } : x))
   627→      )
   628→      toast.success(`${m.name} is now ${next ? 'ACTIVE' : 'INACTIVE'}`)
   629→    } catch (e) {
   630→      toast.error(e instanceof Error ? e.message : 'Failed to update status')
   631→    } finally {
   632→      setRowPending(null)
   633→    }
   634→  }
   635→
   636→  return (
   637→    <div className="space-y-4">
   638→      {/* Toolbar */}
   639→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   640→        <div className="relative flex-1 sm:max-w-xs">
   641→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   642→          <Input
   643→            value={search}
   644→            onChange={(e) => setSearch(e.target.value)}
   645→            placeholder="Search name, generic or brand"
   646→            className="pl-8"
   647→          />
   648→        </div>
   649→        <div className="sm:ml-auto">
   650→          <Button
   651→            onClick={() => {
   652→              setEditing(null)
   653→              setDialogOpen(true)
   654→            }}
   655→          >
   656→            <Plus className="h-4 w-4" /> Add Medicine
   657→          </Button>
   658→        </div>
   659→      </div>
   660→
   661→      {/* Table */}
   662→      <Card className="gap-0 py-4">
   663→        <CardContent className="px-4">
   664→          <div className="overflow-x-auto scrollbar-thin">
   665→            <Table className="min-w-[880px]">
   666→              <TableHeader>
   667→                <TableRow>
   668→                  <TableHead>Medicine</TableHead>
   669→                  <TableHead>Category</TableHead>
   670→                  <TableHead className="text-right">Price</TableHead>
   671→                  <TableHead className="text-center">Stock</TableHead>
   672→                  <TableHead className="text-center">Rx</TableHead>
   673→                  <TableHead className="text-center">Active</TableHead>
   674→                  <TableHead>Expiry</TableHead>
   675→                  <TableHead className="text-right">Edit</TableHead>
   676→                </TableRow>
   677→              </TableHeader>
   678→              <TableBody>
   679→                {loading ? (
   680→                  [...Array(6)].map((_, i) => (
   681→                    <TableRow key={`sk-${i}`}>
   682→                      <TableCell colSpan={8}>
   683→                        <Skeleton className="h-10 w-full" />
   684→                      </TableCell>
   685→                    </TableRow>
   686→                  ))
   687→                ) : filtered.length === 0 ? (
   688→                  <TableRow>
   689→                    <TableCell colSpan={8}>
   690→                      <div className="flex flex-col items-center gap-2 py-10 text-center">
   691→                        <PackageSearch className="h-8 w-8 text-muted-foreground/50" />
   692→                        <p className="text-sm text-muted-foreground">No medicines found.</p>
   693→                      </div>
   694→                    </TableCell>
   695→                  </TableRow>
   696→                ) : (
   697→                  filtered.map((m) => {
   698→                    const expired = m.expiryDate ? new Date(m.expiryDate) < new Date() : false
   699→                    return (
   700→                      <TableRow key={m.id} className="transition-colors hover:bg-accent/40">
   701→                        <TableCell>
   702→                          <div className="flex items-center gap-2.5">
   703→                            <MedImage src={m.image} alt={m.name} className="h-10 w-10 rounded" />
   704→                            <div className="min-w-0">
   705→                              <p className="max-w-52 truncate text-sm font-medium">{m.name}</p>
   706→                              {m.genericName && (
   707→                                <p className="max-w-52 truncate text-xs text-muted-foreground">
   708→                                  {m.genericName}
   709→                                </p>
   710→                              )}
   711→                            </div>
   712→                          </div>
   713→                        </TableCell>
   714→                        <TableCell>
   715→                          <Badge variant="secondary">{m.category?.name ?? 'Uncategorized'}</Badge>
   716→                        </TableCell>
   717→                        <TableCell className="whitespace-nowrap text-right">
   718→                          <span className="font-medium">{fmtBDT(effectivePrice(m))}</span>
   719→                          {m.discountPrice && m.discountPrice < m.price ? (
   720→                            <span className="ml-1.5 text-xs text-muted-foreground line-through">
   721→                              {fmtBDT(m.price)}
   722→                            </span>
   723→                          ) : null}
   724→                        </TableCell>
   725→                        <TableCell>
   726→                          <div className="flex items-center justify-center gap-1">
   727→                            <Button
   728→                              variant="outline"
   729→                              size="icon"
   730→                              className="h-6 w-6"
   731→                              disabled={rowPending === m.id || m.stock <= 0}
   732→                              onClick={() => void adjustStock(m, -1)}
   733→                              title={`Decrease stock of ${m.name}`}
   734→                              aria-label={`Decrease stock of ${m.name}`}
   735→                            >
   736→                              <Minus className="h-3 w-3" />
   737→                            </Button>
   738→                            <span className="w-8 text-center text-sm font-medium">{m.stock}</span>
   739→                            <Button
   740→                              variant="outline"
   741→                              size="icon"
   742→                              className="h-6 w-6"
   743→                              disabled={rowPending === m.id}
   744→                              onClick={() => void adjustStock(m, 1)}
   745→                              title={`Increase stock of ${m.name}`}
   746→                              aria-label={`Increase stock of ${m.name}`}
   747→                            >
   748→                              <Plus className="h-3 w-3" />
   749→                            </Button>
   750→                          </div>
   751→                          <div className="mt-1 flex justify-center">
   752→                            <StockChip stock={m.stock} />
   753→                          </div>
   754→                        </TableCell>
   755→                        <TableCell className="text-center">
   756→                          {m.requiresPrescription ? (
   757→                            <Badge variant="outline" className="border-amber-300 bg-amber-50 text-amber-800">
   758→                              Rx
   759→                            </Badge>
   760→                          ) : (
   761→                            <span className="text-xs text-muted-foreground">—</span>
   762→                          )}
   763→                        </TableCell>
   764→                        <TableCell className="text-center">
   765→                          <Switch
   766→                            checked={m.status === 'ACTIVE'}
   767→                            disabled={rowPending === m.id}
   768→                            onCheckedChange={(v) => void toggleStatus(m, v)}
   769→                            aria-label={`Toggle ${m.name}`}
   770→                          />
   771→                        </TableCell>
   772→                        <TableCell
   773→                          className={`whitespace-nowrap text-xs ${expired ? 'font-medium text-red-600' : 'text-muted-foreground'}`}
   774→                        >
   775→                          {expired ? 'Expired ' : ''}
   776→                          {fmtDate(m.expiryDate)}
   777→                        </TableCell>
   778→                        <TableCell className="text-right">
   779→                          <Button
   780→                            variant="outline"
   781→                            size="icon"
   782→                            className="h-8 w-8"
   783→                            onClick={() => {
   784→                              setEditing(m)
   785→                              setDialogOpen(true)
   786→                            }}
   787→                            title={`Edit ${m.name}`}
   788→                            aria-label={`Edit ${m.name}`}
   789→                          >
   790→                            <Pencil className="h-3.5 w-3.5" />
   791→                          </Button>
   792→                        </TableCell>
   793→                      </TableRow>
   794→                    )
   795→                  })
   796→                )}
   797→              </TableBody>
   798→            </Table>
   799→          </div>
   800→        </CardContent>
   801→      </Card>
   802→
   803→      <MedicineDialog
   804→        open={dialogOpen}
   805→        onOpenChange={setDialogOpen}
   806→        categories={categories}
   807→        initial={editing}
   808→        onSaved={() => void load()}
   809→      />
   810→    </div>
   811→  )
